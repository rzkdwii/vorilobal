-- v1.0.54: budget Lucky Spins per akun dan per fase, persisten saat rejoin.
-- Tidak mengubah fungsi spin, Pity, atau mekanisme rollback yang sudah ada.
local function makePersistentLuckyBudget(userId, phase, maximum)
    local TP = game:GetService("TeleportService")
    local env = (typeof(getgenv) == "function" and getgenv()) or _G
    local key = "VL_Lucky30_Budget_v166_" .. phase .. "_" .. tostring(userId)
    local filename = key .. ".txt"
    -- v1.0.66: limit disimpan per fase agar jatah tidak berubah saat rejoin.
    -- Format penyimpanan lama (v1) masih dapat dibaca.
    local state = {spent = 0, last = nil, rollbackPending = false, limit = nil}

    local function encode()
        return "v2|" .. tostring(state.spent) .. "|"
            .. tostring(state.last or -1) .. "|"
            .. (state.rollbackPending and "1" or "0") .. "|"
            .. tostring(state.limit or -1)
    end
    local function decode(value)
        if type(value) ~= "string" then return false end
        local used, last, pending, limit = value:match(
            "^v2|(%d+)|([%-]?%d+)|([01])|([%-]?%d+)$")
        local legacy = false
        if not used then
            used, last, pending = value:match("^v1|(%d+)|([%-]?%d+)|([01])$")
            legacy = true
        end
        used, last, limit = tonumber(used), tonumber(last), tonumber(limit)
        if not used or used < 0 or used > 100000
            or not last or last < -1 or last > 1000000
            or (not legacy and (not limit or limit < -1 or limit > maximum)) then
            return false
        end
        state.spent = used
        state.last = last
        if last == -1 then state.last = nil end
        state.rollbackPending = pending == "1"
        -- Budget lama yang telah terpakai dipertahankan sampai rollback.
        state.limit = legacy and (used > 0 and maximum or nil)
            or (limit == -1 and nil or limit)
        return true
    end
    local loaded = false
    if typeof(readfile) == "function" then
        local ok, data = pcall(function()
            if typeof(isfile) == "function" and not isfile(filename) then return nil end
            return readfile(filename)
        end)
        if ok then loaded = decode(data) end
    end
    if not loaded then
        local ok, data = pcall(function() return TP:GetTeleportSetting(key) end)
        if ok then loaded = decode(data) end
    end
    if not loaded then loaded = decode(env[key]) end

    local function save()
        local encoded = encode()
        env[key] = encoded
        local fileOK = false
        if typeof(writefile) == "function" then
            fileOK = pcall(function() writefile(filename, encoded) end)
        end
        local tpOK = pcall(function() TP:SetTeleportSetting(key, encoded) end)
        if not fileOK and not tpOK then
            warn("[LUCKY BUDGET " .. phase .. "] Tidak bisa menyimpan budget "
                .. "lintas rejoin; periksa writefile/TeleportSetting")
        end
    end
    local function observe(balance)
        if type(balance) ~= "number" or balance < 0 then
            return state.spent, nil
        end
        local event = nil
        if state.last == nil then
            state.last = balance
            save()
            event = "INIT"
        elseif balance ~= state.last then
            if state.rollbackPending and balance > state.last then
                -- Rollback berhasil: hitungan dan jatah diulang dari saldo pulih.
                state.spent = 0
                state.limit = nil
                state.rollbackPending = false
                event = "RESET_AFTER_ROLLBACK"
            elseif balance < state.last then
                state.spent = state.spent + (state.last - balance)
                event = "SPENT"
            else
                -- Tambahan saldo tanpa rollback bukan alasan menghapus riwayat.
                event = "BALANCE_INCREASE"
            end
            state.last = balance
            save()
        end
        if state.limit == nil and balance > 0 then
            -- Jangan mengunci jatah saat saldo sementara terbaca 0
            -- sebelum StyleController selesai sinkron.
            -- Jangan menghitung ulang pembagian jika rejoin di tengah fase.
            if state.spent > 0 then
                state.limit = maximum -- pemulihan budget lawas yang sedang berjalan
            elseif phase == "TIMESKIP_OKAZU_FIRST" and balance <= 29 then
                state.limit = math.ceil(balance / 2)
            elseif phase == "THE_TWINS" and balance <= 29 then
                -- Semua sisa Timeskip Okazu menjadi jatah The Twins, tanpa dibagi lagi.
                state.limit = balance
            else
                state.limit = maximum
            end
            save()
            event = event or "LIMIT_INIT"
        end
        return state.spent, event
    end
    local function markRollback()
        state.rollbackPending = true
        save()
    end
    local function resetAfterSuccess()
        state.spent = 0
        state.last = nil
        state.limit = nil
        state.rollbackPending = false
        save()
    end
    return {
        observe = observe,
        spent = function() return state.spent end,
        limit = function() return state.limit or maximum end,
        capped = function() return state.spent >= (state.limit or maximum) end,
        markRollback = markRollback,
        resetAfterSuccess = resetAfterSuccess,
        isRollbackPending = function() return state.rollbackPending end,
        maximum = maximum,
        loaded = loaded
    }
end

                local function initialSlotRank(id)
                    if type(id) ~= "string" or id == "" then
                        return nil, "slot belum siap"
                    end
                    local def = styleDef(id)
                    local dn = def and lower(def.DisplayName) or ""
                    local base = lower(id)
                    -- Target Timeskip Okazu/The Twins dan Mamori mutlak dilindungi.
                    if base:find("mamori", 1, true) or base:find("timeskip okazu", 1, true)
                        or dn == "mamori" or dn == "timeskip okazu" or dn == "the twins"
                        or base:find("twin1_5", 1, true)
                        or base:find("timeskiptoru4", 1, true) then
                        return nil, "target dilindungi"
                    end
                    -- Dicek SEBELUM knownProtected (keduanya ada di sana).
                    if exceptionNames[base] or exceptionNames[dn] then
                        return 6, "fallback terakhir (Rintaro/Kimiro)"
                    end
                    if commonNames[base] or commonNames[dn] then return 1, "Common" end
                    if knownProtected[base] or knownProtected[dn] then
                        return nil, "Secret/Ultra/Evo dilindungi"
                    end
                    if not def then return nil, "data rarity tidak tersedia" end
                    local rarity = def.Rarity
                    if type(rarity) == "string" then
                        local r = lower(rarity)
                        if r == "secret" or r == "ultra" or r == "evo" or r == "evolution" then
                            return nil, "rarity " .. r .. " dilindungi"
                        end
                        if r == "common" or r == "uncommon" then return 1, r end
                        if r == "rare" then return 2, r end
                        if r == "legendary" then return 3, r end
                        if r == "godly" then return 4, r end
                        return nil, "rarity string tidak dikenal: " .. r
                    end
                    if type(rarity) == "number" then
                        if not calibrationComplete then calibrateProtectedRarity() end
                        if rarityProtectedNumbers[rarity] ~= nil then
                            return nil, "rarity numeric " .. tostring(rarity)
                                .. " dilindungi (" .. rarityProtectedNumbers[rarity] .. ")"
                        end
                        if calibrationComplete then
                            return 5, "non-Secret/Ultra/Evo (R=" .. tostring(rarity) .. ")"
                        end
                        return nil, "kalibrasi rarity belum lengkap"
                    end
                    return nil, "rarity belum terbaca"
                end
                local function slotSummary(slots)
                    if type(slots) ~= "table" then return "inventori belum tersedia" end
                    local parts = {}
                    for key, id in pairs(slots) do
                        local index = tonumber(key)
                        if index and index % 1 == 0 then
                            local _, label = initialSlotRank(id)
                            parts[#parts+1] = { index=index, text=tostring(index)
                                .. ":" .. tostring(id) .. " (" .. tostring(label) .. ")" }
                        end
                    end
                    table.sort(parts, function(x,y) return x.index < y.index end)
                    local result = {}
                    for i=1,math.min(#parts,8) do result[#result+1] = parts[i].text end
                    return table.concat(result, "; ")
                end

                local function twinsSlotAvailability()
                    local slots = readSignal(teamA and teamA.Styles)
                    if type(slots) ~= "table" then
                        return nil, "data slot belum tersedia"
                    end
                    local knownSlots, uncertain = 0, 0
                    for key, id in pairs(slots) do
                        local idx = tonumber(key)
                        if idx and idx >= 1 and idx % 1 == 0 then
                            if type(id) ~= "string" or id == "" then
                                uncertain += 1
                            else
                                knownSlots += 1
                                local rank, reason = initialSlotRank(id)
                                if rank then
                                    return true, "slot " .. tostring(idx) .. " (" .. tostring(reason) .. ")"
                                end
                                -- Hanya 'dilindungi' yang membuktikan slot tidak aman.
                                -- Rarity belum terbaca TIDAK dianggap slot habis.
                                if type(reason) ~= "string"
                                    or not reason:find("dilindungi", 1, true) then
                                    uncertain += 1
                                end
                            end
                        end
                    end
                    if knownSlots == 0 or uncertain > 0 then
                        return nil, "slot/rarity belum pasti (terbaca="
                            .. tostring(knownSlots) .. ", belum siap=" .. tostring(uncertain) .. ")"
                    end
                    return false, "semua slot terlindungi: " .. slotSummary(slots)
                end

                local function describeResult(x, depth)
                    depth = depth or 0
                    if type(x) == "table" then
                        if depth >= 2 then return "{...}" end
                        local parts, n = {}, 0
                        for k, v in pairs(x) do
                            n += 1
                            if n > 8 then parts[#parts+1] = "..."; break end
                            parts[#parts+1] = tostring(k) .. "=" .. describeResult(v, depth + 1)
                        end
                        return "{" .. table.concat(parts, ",") .. "}"
                    end
                    return tostring(x)

local function requestRejoin()
    local guardKey = "__VL_MAMORI_REJOIN_GUARD_" .. tostring(Player.UserId)
    if ENV[guardKey] == tostring(game.JobId) then
        __mamoriTrace("REJOIN sedang diawasi di server ini")
        return true
    end
    ENV[guardKey] = tostring(game.JobId)
    task.spawn(function()
        local HttpService = game:GetService("HttpService")
        local currentJob = tostring(game.JobId)
        local begun, failed = false, false
        local failDescription = ""
        local onTP = Player.OnTeleport:Connect(function(state)
            __mamoriTrace("TELEPORT STATE=" .. tostring(state))
            if state == Enum.TeleportState.Started or state == Enum.TeleportState.InProgress then
                begun = true
            end
        end)
        local onFail = TP.TeleportInitFailed:Connect(function(p, result, msg, placeId)
            if p == Player then
                failed = true
                failDescription = tostring(result).." / "..tostring(msg)
                __mamoriTrace("TELEPORT GAGAL="..failDescription.." place="..tostring(placeId))
            end
        end)
        local function waitForTeleport(seconds)
            local deadline = os.clock() + seconds
            while os.clock() < deadline do
                if begun then return true end
                if failed then return false end
                task.wait(0.2)
            end
            return begun
        end
        local function getCandidates()
            local out, cursor, seen = {}, nil, {}
            for pageNum = 1, 4 do
                local url = "https://games.roblox.com/v1/games/" .. tostring(game.PlaceId)
                    .. "/servers/Public?sortOrder=Asc&limit=100"
                if cursor and cursor ~= "" then
                    url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
                end
                local ok, decoded = pcall(function()
                    return HttpService:JSONDecode(game:HttpGet(url))
                end)
                if not ok or type(decoded) ~= "table" then
                    __mamoriTrace("SERVER LIST gagal page="..pageNum.." : "..tostring(decoded):sub(1,100))
                    break
                end
                for _, server in ipairs(decoded.data or {}) do
                    if type(server.id) == "string" and server.id ~= currentJob
                        and not seen[server.id]
                        and (tonumber(server.playing) or math.huge) < (tonumber(server.maxPlayers) or 0) then
                        seen[server.id] = true
                        out[#out+1] = {id=server.id, count=tonumber(server.playing) or 99}
                    end
                end
                cursor = decoded.nextPageCursor
                if not cursor or #out >= 25 then break end
            end
            table.sort(out, function(a,b)
                if a.count == b.count then return a.id < b.id end
                return a.count < b.count
            end)
            return out
        end
        __mamoriTrace("REJOIN SAME-PLACE | source="..tostring(game.PlaceId).." goal="..tostring(game.PlaceId))
        local candidates = getCandidates()
        __mamoriTrace("SERVER PUBLIK PLACE INI ditemukan="..#candidates)
        local total = math.min(#candidates, 4)
        for i = 1, total do
            if begun then break end
            failed = false
            local target = candidates[i].id
            __mamoriTrace("SERVER HOP "..i.." -> "..target)
            local ok, err = pcall(function()
                TP:TeleportToPlaceInstance(game.PlaceId, target, Player)
            end)
            __mamoriTrace("HOP REQUEST="..tostring(ok)..(ok and "" or " err="..tostring(err)))
            if ok and waitForTeleport(9) then break end
            task.wait(1)
        end
        if not begun then
            failed = false
            __mamoriTrace("FALLBACK: Teleport ke game.PlaceId")
            local ok, err = pcall(function() TP:Teleport(game.PlaceId, Player) end)
            __mamoriTrace("FALLBACK REQUEST="..tostring(ok)..(ok and "" or " err="..tostring(err)))
            if ok then waitForTeleport(12) end
        end
        if not begun then
            __mamoriTrace("REJOIN TERTAHAN; menunggu pindah ke server lain")
            __mamoriTrace("KETERANGAN="..failDescription)
            __mamoriTrace("Kirim log teleport; tidak ada Kick")
        else
            __mamoriTrace("TELEPORT MULAI; menunggu sesi server baru")
        end
        pcall(function() onTP:Disconnect() end)
        pcall(function() onFail:Disconnect() end)
        ENV[guardKey] = nil
    end)
    return true
end

local function waitForStableTutorialSkip(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds
    local stableSince = nil
    local lastButton = nil

    while __xynIsCurrentRun() and os.clock() < deadline do
        local button = tutorialMainSkip()

        if button then
            if button ~= lastButton then
                lastButton = button
                stableSince = os.clock()
            elseif stableSince and os.clock() - stableSince >= 0.45 then
                return button
            end
        else
            lastButton = nil
            stableSince = nil
        end

        task.wait(0.10)
    end

    return nil
end

local function waitForTutorialPrompt(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds

    while __xynIsCurrentRun() and os.clock() < deadline do
        local option1, option2, prompt = tutorialPromptOptions()

        if prompt and tutorialVisible(prompt) then
            return option1, option2, prompt
        end

        task.wait(0.05)
    end

    return nil, nil, nil
end

local function waitForTutorialEnd(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds
    local clearedAt = nil

    while __xynIsCurrentRun() and os.clock() < deadline do
        if not tutorialStillActive() then
            clearedAt = clearedAt or os.clock()
            -- Jangan anggap sukses jika tombol/GUI hilang sesaat saat transisi.
            if os.clock() - clearedAt >= 0.7 then
                return true
            end
        else
            clearedAt = nil
        end
        task.wait(0.12)
    end

    return false
end

local function runAutoSkipTutorialGate()
    -- Klik tengah dahulu agar tutorial/Coach bisa menampilkan tombol skip.
    task.wait(0.5)
    clickTutorialCenter()

    local detectDeadline = os.clock() + 4
    while __xynIsCurrentRun() and os.clock() < detectDeadline do
        if tutorialModeDetected() then break end
        task.wait(0.12)
    end
    if not __xynIsCurrentRun() then return true, false end
    if not tutorialModeDetected() then return false, true end

    local attempts = 0
    while __xynIsCurrentRun() do
        if waitForTutorialEnd(1.0) then
            return true, true
        end

        attempts += 1
        -- Gagal? Klik tengah lagi, lalu coba Skip lagi, tanpa pindah server.
        clickTutorialCenter()
        task.wait(0.6)
        if waitForTutorialEnd(1.0) then return true, true end

        local mainButton = waitForStableTutorialSkip(4)
        if mainButton then
            pressTutorialButton(mainButton)
            local option1, option2, prompt = waitForTutorialPrompt(3)
            if prompt and tutorialVisible(prompt) then
                pressTutorialButton(option1)
                if waitForTutorialEnd(2) then return true, true end

                -- Coba tombol skip bawaan lagi lalu Option2 seperti patch asli.
                mainButton = waitForStableTutorialSkip(4)
                if mainButton then
                    pressTutorialButton(mainButton)
                    option1, option2, prompt = waitForTutorialPrompt(3)
                    if prompt and tutorialVisible(prompt) then
                        pressTutorialButton(option2)
                        if waitForTutorialEnd(2) then return true, true end
                    end
                end
            end
        end

        if waitForTutorialEnd(1.0) then return true, true end
        if attempts % 3 == 0 then
            warn("[STARTUP FIX] Tutorial belum selesai; ulang klik tengah + Skip, percobaan " .. attempts)
        end
        task.wait(0.8)
    end

    return true, false
end

local BAD_SERVER_FILE = "xyn_bad_servers.json"
local BAD_SERVER_TELEPORT_KEY = "__xynBadServerHistory"
local MAX_BAD_SERVERS = 50
local SERVER_PAGE_LIMIT = 100
local SERVER_MAX_PAGES = 6
local SERVER_FETCH_RETRIES = 4

local badServerOrder = {}
local badServerSet = {}
local rejoinStarted = false

local function normalizeJobId(value)
    if type(value) ~= "string" then
        return nil
    end
    value = value:match("^%s*(.-)%s*$")
    if value == "" then
        return nil
    end
    return value
end

local function rememberBadServer(jobId)
    jobId = normalizeJobId(jobId)
    if not jobId then
        return false
    end

    if badServerSet[jobId] then
        -- Pindahkan ke posisi terbaru agar trimming menyimpan yang paling relevan.
        for i = #badServerOrder, 1, -1 do
            if badServerOrder[i] == jobId then
                table.remove(badServerOrder, i)
                break
            end
        end
    else
        badServerSet[jobId] = true
    end

    table.insert(badServerOrder, jobId)

    while #badServerOrder > MAX_BAD_SERVERS do
        local removed = table.remove(badServerOrder, 1)
        if removed then
            badServerSet[removed] = nil
        end
    end

    return true
end

local function mergeBadServerList(value)
    if type(value) ~= "table" then
        return
    end

    for _, jobId in ipairs(value) do
        rememberBadServer(jobId)
    end
end

local function decodeBadServerList(raw)
    if type(raw) ~= "string" or raw == "" then
        return nil
    end

    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)

    if ok and type(decoded) == "table" then
        return decoded
    end

    return nil
end

local function loadBadServers()
    -- 1) TeleportSetting: berguna saat pindah server dalam client yang sama.
    pcall(function()
        local raw = TeleportService:GetTeleportSetting(BAD_SERVER_TELEPORT_KEY)
        if type(raw) == "table" then
            mergeBadServerList(raw)
        elseif type(raw) == "string" then
            mergeBadServerList(decodeBadServerList(raw))
        end
    end)

    -- 2) File executor: fallback/persistence jika Delta mendukung readfile/writefile.
    if typeof(readfile) == "function" then
        local ok, raw = pcall(function()
            if typeof(isfile) == "function" and not isfile(BAD_SERVER_FILE) then
                return nil
            end
            return readfile(BAD_SERVER_FILE)
        end)

        if ok and type(raw) == "string" then
            mergeBadServerList(decodeBadServerList(raw))
        end
    end
end

local function persistBadServers()
    local encoded = nil
    pcall(function()
        encoded = HttpService:JSONEncode(badServerOrder)
    end)

    if type(encoded) ~= "string" then
        return false
    end

    pcall(function()
        TeleportService:SetTeleportSetting(BAD_SERVER_TELEPORT_KEY, encoded)
    end)

    if typeof(writefile) == "function" then
        pcall(function()
            writefile(BAD_SERVER_FILE, encoded)
        end)
    end

    return true
end

loadBadServers()

local function isBadServer(jobId)
    jobId = normalizeJobId(jobId)
    return jobId ~= nil and badServerSet[jobId] == true
end

local function stableStringHash(value)
    value = tostring(value or "")
    local h = 17
    for i = 1, #value do
        h = (h * 131 + string.byte(value, i)) % 2147483647
    end
    return h
end

local function fetchPublicServerPage(cursor)
    local url = "https://games.roblox.com/v1/games/"
        .. tostring(game.PlaceId)
        .. "/servers/Public?sortOrder=Asc&limit="
        .. tostring(SERVER_PAGE_LIMIT)

    if cursor and cursor ~= "" then
        url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
    end

    local okHttp, body = pcall(function()
        return game:HttpGet(url)
    end)

    if not okHttp or type(body) ~= "string" or body == "" then
        return nil
    end

    local okDecode, decoded = pcall(function()
        return HttpService:JSONDecode(body)
    end)

    if not okDecode or type(decoded) ~= "table" then
        return nil
    end

    return decoded
end

local function collectServerCandidates()
    local candidates = {}
    local seen = {}
    local cursor = nil
    local thisJob = currentJobId()

    for _ = 1, SERVER_MAX_PAGES do
        if not detectorIsCurrent() then
            break
        end

        local page = nil
        for retry = 1, SERVER_FETCH_RETRIES do
            page = fetchPublicServerPage(cursor)
            if page then
                break
            end
            task.wait(0.35 * retry)
        end

        if not page then
            break
        end

        if type(page.data) == "table" then
            for _, server in ipairs(page.data) do
                local id = normalizeJobId(server.id)
                local playing = tonumber(server.playing) or math.huge
                local maxPlayers = tonumber(server.maxPlayers) or 0

                if id
                    and id ~= thisJob
                    and not badServerSet[id]
                    and not seen[id]
                    and maxPlayers > 0
                    and playing < maxPlayers
                then
                    seen[id] = true
                    table.insert(candidates, {
                        id = id,
                        playing = playing,
                        maxPlayers = maxPlayers,
                    })
                end
            end
        end

        cursor = page.nextPageCursor
        if type(cursor) ~= "string" or cursor == "" then
            break
        end

        -- Biasanya satu-dua halaman sudah cukup. Jangan spam API kalau kandidat banyak.
        if #candidates >= 40 then
            break
        end
    end

    return candidates
end

local function chooseServerCandidate(candidates)
    if type(candidates) ~= "table" or #candidates == 0 then
        return nil
    end

    -- Sort dulu supaya semua client punya basis yang sama.
    table.sort(candidates, function(a, b)
        if a.playing == b.playing then
            return a.id < b.id
        end
        return a.playing < b.playing
    end)

    -- Sebarkan akun script ke server berbeda. UserId membuat index berbeda
    -- sehingga beberapa account yang hop bersamaan tidak mudah berkumpul lagi.
    local salt = stableStringHash(currentJobId())
    local uid = tonumber(player.UserId) or 0
    local index = ((uid + salt) % #candidates) + 1
    return candidates[index]
end

local function tryTeleportToServer(jobId)
    jobId = normalizeJobId(jobId)
    if not jobId or jobId == currentJobId() or badServerSet[jobId] then
        return false
    end

    queueSelfForNextTeleport()
    persistBadServers()

    local ok = pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, jobId, player)
    end)

    return ok
end

local function serverHopAway(reason)
    -- Tidak ada teleport/server-hop saat tutorial masih aktif.
    if tutorialStillActive() then
        warn("[STARTUP FIX] Server hop ditahan karena tutorial aktif: " .. tostring(reason))
        return false
    end
    if rejoinStarted or not detectorIsCurrent() then
        return false
    end

    rejoinStarted = true

    local oldJob = currentJobId()
    if oldJob then
        rememberBadServer(oldJob)
    end
    persistBadServers()

    -- Coba beberapa kali mengambil server publik yang berbeda.
    for attempt = 1, 5 do
        if not detectorIsCurrent() then
            return false
        end

        local candidates = collectServerCandidates()
        local chosen = chooseServerCandidate(candidates)

        if chosen and tryTeleportToServer(chosen.id) then
            -- Jika request teleport gagal secara asynchronous dan kita masih berada
            -- di server ini beberapa detik kemudian, buka lock agar bisa retry.
            task.delay(10, function()
                if detectorIsCurrent()
                    and rejoinStarted
                    and currentJobId() == oldJob
                then
                    rejoinStarted = false
                    task.spawn(function()
                        serverHopAway(reason or "retry")
                    end)
                end
            end)
            return true
        end

        task.wait(0.6 * attempt)
    end

    -- Last resort: regular teleport. Server lama sudah berada di blacklist;
    -- kalau Roblox mengirim kita kembali ke JobId yang sama, startup detector
    -- akan langsung hop lagi.
    queueSelfForNextTeleport()
    persistBadServers()

    local ok = pcall(function()
        TeleportService:Teleport(game.PlaceId, player)
    end)

    if not ok then
        rejoinStarted = false
        return false
    end

    task.delay(10, function()
        if detectorIsCurrent()
            and rejoinStarted
            and currentJobId() == oldJob
        then
            rejoinStarted = false
            task.spawn(function()
                serverHopAway(reason or "retry")
            end)
        end
    end)

    return true
end

local function getStartupAveragePingMs()
    local total = 0
    local samples = 0
    for _ = 1, 4 do
        if not detectorIsCurrent() then
            return nil
        end
        local pingMs = getNetworkPingMs()
        if pingMs then
            total += pingMs
            samples += 1
        end
        task.wait(0.25)
    end
    if samples == 0 then
        return nil
    end
    return total / samples
end

    serverHopAway("high_ping")
end

local function runStartupDetection()
    task.wait(1)
    if not detectorIsCurrent() then
        return
    end

    -- Kalau Roblox menaruh kita kembali ke server yang pernah ditandai buruk,
    -- jangan tunggu kondisi lain: langsung hop ke JobId berbeda.
    local jobId = currentJobId()
    if jobId and isBadServer(jobId) then
        serverHopAway("returned_to_bad_server")
        return
    end

    local pingMs = getStartupAveragePingMs()
    if pingMs and pingMs > PING_LIMIT_MS then
        handleHighPing()
        return
    end
end

local knownTeamObjects = {
    [1] = nil,
    [2] = nil,
}
local lastRequestedTeamId = nil
local lastRequestAt = 0
local function rememberTeamObject(teamId, teamObject)
    if not teamObject or not teamObject:IsA("Team") then
        return
    end
    local otherId = teamId == 1 and 2 or 1
    if knownTeamObjects[otherId] == teamObject then
        return
    end
    knownTeamObjects[teamId] = teamObject
end
local function parseTeamIdValue(value)
    if typeof(value) == "number" then
        local n = tonumber(value)
        if n == 1 or n == 2 then
            return n
        end
        return nil
    end
    if typeof(value) == "string" then
        local lower = string.lower(value)
        local direct = tonumber(lower)
        if direct == 1 or direct == 2 then
            return direct
        end
        local fromName = lower:match("team%s*(%d+)") or lower:match("team(%d+)")
        fromName = tonumber(fromName)
        if fromName == 1 or fromName == 2 then
            return fromName
        end
    end
    return nil
end
local function extractTeamIdFromRemoteResult(result)
    local direct = parseTeamIdValue(result)
    if direct then
        return direct
    end
    if typeof(result) == "table" then
        for _, value in ipairs({
            result.TeamId,
            result.TeamID,
            result.teamId,
            result.teamID,
            result.Team,
            result.team,
            result.Id,
            result.id,
        }) do
            local parsed = parseTeamIdValue(value)
            if parsed then
                return parsed
            end
        end
    end
    return nil
end
local function joinTeam(teamId)
    if not refreshGameRemotes() then
        return false, nil
    end
    lastRequestedTeamId = teamId
    lastRequestAt = os.clock()
    local beforeTeam = player.Team
    local requestResult = nil
    local ok = pcall(function()
        requestResult = RequestJoin:InvokeServer(teamId, 5)
        Serve:InvokeServer(Vector3.zero, 0)
    end)
    local afterTeam = player.Team
    if ok and afterTeam and afterTeam ~= beforeTeam then
        rememberTeamObject(teamId, afterTeam)
    end
    local resultTeamId = extractTeamIdFromRemoteResult(requestResult)
    if ok and resultTeamId == teamId and afterTeam then
        rememberTeamObject(teamId, afterTeam)
    end
    return ok, requestResult
end
local function normalizeTeamValue(value)
    if value == nil then return nil end
    return string.lower(tostring(value)):gsub("[^%w]", "")
end
local function valueMatchesTeam(value, teamId)
    local normalized = normalizeTeamValue(value)
    if not normalized then return false end
    return normalized == tostring(teamId)
        or normalized == ("team" .. tostring(teamId))
end
local function refreshKnownTeamObjectsFromService()
    local serviceTeams = TeamsService:GetTeams()
    for _, teamObject in ipairs(serviceTeams) do
        for teamId = 1, 2 do
            local attr = teamObject:GetAttribute("TeamId")
                or teamObject:GetAttribute("TeamID")
                or teamObject:GetAttribute("Id")
            if valueMatchesTeam(teamObject.Name, teamId)
                or valueMatchesTeam(attr, teamId)
            then
                rememberTeamObject(teamId, teamObject)
            end
        end
    end
    if #serviceTeams == 2 then
        if knownTeamObjects[1] and not knownTeamObjects[2] then
            for _, teamObject in ipairs(serviceTeams) do
                if teamObject ~= knownTeamObjects[1] then
                    rememberTeamObject(2, teamObject)
                end
            end
        elseif knownTeamObjects[2] and not knownTeamObjects[1] then
            for _, teamObject in ipairs(serviceTeams) do
                if teamObject ~= knownTeamObjects[2] then
                    rememberTeamObject(1, teamObject)
                end
            end
        end
    end
end
local function instanceTeamMatches(instance, teamId)
    if not instance then return false end
    if knownTeamObjects[teamId] and instance == knownTeamObjects[teamId] then
        return true
    end
    if valueMatchesTeam(instance.Name, teamId) then
        return true
    end
    local attrs = {
        instance:GetAttribute("TeamId"),
        instance:GetAttribute("TeamID"),
        instance:GetAttribute("teamId"),
        instance:GetAttribute("Team"),
        instance:GetAttribute("team"),
    }
    for _, value in ipairs(attrs) do
        if valueMatchesTeam(value, teamId) then
            return true
        end
    end
    return false
end
local function containerTeamMatches(container, teamId)
    if not container then return false end
    if instanceTeamMatches(container, teamId) then
        return true
    end
    for _, childName in ipairs({"Team", "TeamId", "TeamID", "team", "teamId"}) do
        local child = container:FindFirstChild(childName)
        if child and child:IsA("ValueBase") and valueMatchesTeam(child.Value, teamId) then
            return true
        end
    end
    return false
end
local function teamLooksConfirmedFor(targetPlayer, teamId)
    if not targetPlayer then return false end

    local teamObject = targetPlayer.Team
    if teamObject and instanceTeamMatches(teamObject, teamId) then
        return true
    end
    if containerTeamMatches(targetPlayer, teamId) then
        return true
    end
    local targetCharacter = targetPlayer.Character
    if targetCharacter and containerTeamMatches(targetCharacter, teamId) then
        return true
    end
    return false
end
local function teamLooksConfirmed(teamId)
    return teamLooksConfirmedFor(player, teamId)
end
local function getCurrentTeamRoster()
    return {player}
end
local function getPresentSignatureFrom(snapshot)
    local ids = {}
    for _, p in ipairs(snapshot) do
        table.insert(ids, tostring(p.UserId))
    end
    return table.concat(ids, "|")
end
local function getPresentSignature()
    return getPresentSignatureFrom(getCurrentTeamRoster())
end
local function getTargetTeamFromSnapshot(targetPlayer, snapshot)
    if not targetPlayer then return nil, nil end
    for index, candidate in ipairs(snapshot) do
        if candidate == targetPlayer or candidate.UserId == targetPlayer.UserId then
            -- Pilihan stabil per akun, tanpa daftar nama tetap.
            local teamId = (math.abs(tonumber(targetPlayer.UserId) or 0) % 2) + 1
            return teamId, index
        end
    end
    return nil, nil
end
local TEAM_ROSTER_STABLE_TIME = 1.50
local TEAM_CONFIRM_STABLE_TIME = 0.80
local TEAM_RETRY_DELAY = 0.85
local TEAM_SANITY_INTERVAL = 2.00
local teamGeneration = 0
local teamCycleRunningGeneration = nil
local correctionRunningGeneration = nil
local teamJoinFinished = false
local lockedTeamId = nil
local lockedRosterSignature = nil
local roundResetActive = false
local function generationIsCurrent(generation)
    return getgenv().main
        and detectorIsCurrent()
        and generation == teamGeneration
end
local function waitForStableRoster(generation)
    local lastSignature = nil
    local stableSince = 0
    while generationIsCurrent(generation) do
        local snapshot = getCurrentTeamRoster()
        local signature = getPresentSignatureFrom(snapshot)
        if signature ~= lastSignature then
            lastSignature = signature
            stableSince = os.clock()
        elseif os.clock() - stableSince >= TEAM_ROSTER_STABLE_TIME then
            return snapshot, signature
        end
        task.wait(0.15)
    end
    return nil, nil
end
local function waitForStableLocalConfirmation(generation, targetTeamId)
    local stableSince = nil
    while generationIsCurrent(generation) do
        if teamLooksConfirmed(targetTeamId) then
            stableSince = stableSince or os.clock()
            if os.clock() - stableSince >= TEAM_CONFIRM_STABLE_TIME then
                return true
            end
        else
            stableSince = nil
            return false
        end
        task.wait(0.10)
    end
    return false
end
local function runTeamCycle(generation)
    local snapshot, signature = waitForStableRoster(generation)
    if not snapshot or not generationIsCurrent(generation) then
        return false
    end
    local targetTeamId, position = getTargetTeamFromSnapshot(player, snapshot)
    if not targetTeamId then
        return false
    end
    lockedTeamId = targetTeamId
    lockedRosterSignature = signature
    teamJoinFinished = false
    if position then
        task.wait(0.12 + ((position - 1) * 0.18))
    end
    local lastTeamLog = -100
    while generationIsCurrent(generation) do
        if teamLooksConfirmed(targetTeamId) then
            if waitForStableLocalConfirmation(generation, targetTeamId) then
                teamJoinFinished = true
                warn('[LEVEL TEAM] Bergabung dan terkonfirmasi: Team ' .. tostring(targetTeamId))
                return true
            end
        end
        local sent = joinTeam(targetTeamId)
        if os.clock() - lastTeamLog >= 12 then
            lastTeamLog = os.clock()
            warn('[LEVEL TEAM] Request Team ' .. tostring(targetTeamId)
                .. ' | remote=' .. tostring(sent)
                .. ' | confirmed=' .. tostring(teamLooksConfirmed(targetTeamId)))
        end
        task.wait(TEAM_RETRY_DELAY)
    end
    return false
end
local function launchTeamCycle()
    if roundResetActive or not getgenv().main then
        return
    end
    local generation = teamGeneration
    if teamCycleRunningGeneration == generation then
        return
    end
    teamCycleRunningGeneration = generation
    task.spawn(function()
        runTeamCycle(generation)
        if teamCycleRunningGeneration == generation then
            teamCycleRunningGeneration = nil
        end
    end)
end
local function invalidateTeamAssignment(startImmediately)
    teamGeneration = teamGeneration + 1
    teamJoinFinished = false
    lockedTeamId = nil
    lockedRosterSignature = nil
    correctionRunningGeneration = nil
    if startImmediately and not roundResetActive then
        launchTeamCycle()
    end
end
local function initializeJoins()
    if teamGeneration == 0 then
        invalidateTeamAssignment(true)
    else
        launchTeamCycle()
    end
    while getgenv().main and detectorIsCurrent() do
        if teamJoinFinished and lockedTeamId then
            return true
        end
        if not roundResetActive and teamCycleRunningGeneration == nil then
            launchTeamCycle()
        end
        task.wait(0.15)
    end
    return false
end
local function correctLockedTeam()
    if roundResetActive then
        return
    end
    if not lockedTeamId then
        launchTeamCycle()
        return
    end
    local generation = teamGeneration
    if correctionRunningGeneration == generation then
        return
    end
    correctionRunningGeneration = generation
    teamJoinFinished = false
    local targetTeamId = lockedTeamId
    task.spawn(function()
        while generationIsCurrent(generation)
            and not roundResetActive
            and lockedTeamId == targetTeamId
        do
            if teamLooksConfirmed(targetTeamId) then
                if waitForStableLocalConfirmation(generation, targetTeamId) then
                    teamJoinFinished = true
                    break
                end
            end
            joinTeam(targetTeamId)
            task.wait(TEAM_RETRY_DELAY)
        end
        if correctionRunningGeneration == generation then
            correctionRunningGeneration = nil
        end
    end)
end
local function sanityCheckTeam()
    if roundResetActive or not getgenv().main then
        return
    end
    -- Team retries must not interrupt Coach/tutorial. Action workers do not wait on this confirmation.
    if not tutorialGateCleared or tutorialStillActive() then
        return
    end
    -- The primary team-cycle already owns retries; avoid a second concurrent RequestJoin worker.
    if teamCycleRunningGeneration == teamGeneration then
        return
    end
    local currentSignature = getPresentSignature()
    if lockedRosterSignature and currentSignature ~= lockedRosterSignature then
        invalidateTeamAssignment(true)
        return
    end
    if not lockedTeamId then
        launchTeamCycle()
        return
    end
    if not teamLooksConfirmed(lockedTeamId) then
        correctLockedTeam()
        return
    end
end
local function beginRoundTeamReset()
    if roundResetActive then return end
    roundResetActive = true
    teamGeneration = teamGeneration + 1
    teamJoinFinished = false
    lockedTeamId = nil
    lockedRosterSignature = nil
    correctionRunningGeneration = nil
end
local function finishRoundTeamReset()
    roundResetActive = false
    invalidateTeamAssignment(true)
end

        warn("[PITY FASE 1] Tombol TimeskipToru4 tidak ditemukan di SecretPity")
        return false
    end
    local ups = (debug and debug.getupvalues) or getupvalues
    if type(ups) ~= "function" or type(getconnections) ~= "function" then
        warn("[PITY FASE 1] getconnections/getupvalues tidak tersedia")
        return false
    end
    local ok, connections = pcall(getconnections, btn.Activated)
    if not ok or type(connections) ~= "table" then return false end
    for _, conn in ipairs(connections) do
        if type(conn.Function) == "function" then
            local good, vals = pcall(ups, conn.Function)
            if good and type(vals) == "table" then
                for _, obj in pairs(vals) do
                    if type(obj) == "table" and type(obj.OnActivated) == "function" then
                        local fired, err = pcall(obj.OnActivated)
                        if not fired then
                            fired, err = pcall(obj.OnActivated, obj)
                        end
                        if not fired then
                            warn("[PITY FASE 1] Callback gagal: " .. tostring(err))
                        else
                            for _ = 1, 25 do
                                if preferredFirstPity() == true then
                                    print("[PITY FASE 1] TimeskipToru4 terkonfirmasi")
                                    return true
                                end
                                task.wait(0.2)
                            end
                            warn("[PITY FASE 1] Callback jalan tapi pilihan belum terkonfirmasi")
                        end
                    end
                end
            end
        end
    end
    return false
end
local function waitForFirstPity()
    local selectedHere = false
    local lastRejoin = -100
    while running and Player.Parent do
        local selected = preferredFirstPity()
        if selected == true then
            if firstPityPending() then
                if selectedHere then
                    if os.clock() - lastRejoin > 14 then
                        lastRejoin = os.clock()
                        print("[PITY FASE 1] Rejoin untuk menyimpan TimeskipToru4; spin ditahan")
                        local ok, err = pcall(function() TeleportService:Teleport(game.PlaceId, Player) end)
                        if not ok then warn("[PITY FASE 1] Rejoin gagal: " .. tostring(err)) end
                    end
                    task.wait(2)
                else
                    clearFirstPityPending()
                    print("[PITY FASE 1] TimeskipToru4 tersimpan setelah rejoin; spin diizinkan")
                    return true
                end
            else
                print("[PITY FASE 1] TimeskipToru4 telah terpilih, spin diizinkan")
                return true
            end
        elseif selected == false then
            if firstPityPending() then clearFirstPityPending() end
            if selectFirstPity() then
                if saveFirstPityPending() then
                    selectedHere = true
                    print("[PITY FASE 1] TimeskipToru4 dipilih; wajib rejoin sebelum spin")
                else
                    warn("[PITY FASE 1] Gagal menyimpan penanda Pity; tahan spin")
                end
            end
            task.wait(2)
        else
            print("[PITY FASE 1] PreferredSecret belum siap; menunggu")
            task.wait(2)
        end
    end
    return false
end
if not waitForFirstPity() then
    warn("[PITY FASE 1] TimeskipToru4 belum siap; AutoSpin tidak dimulai")
    return
end

local function getRollRF()
    local idx = RS:FindFirstChild("Packages")
        and RS.Packages:FindFirstChild("_Index")

    if idx then
        for _, f in ipairs(idx:GetChildren()) do
            local k = f:FindFirstChild("knit")

            if k and k:FindFirstChild("Services") then
                local svc = k.Services:FindFirstChild("StyleService")

                if svc
                    and svc:FindFirstChild("RF")
                    and svc.RF:FindFirstChild("Roll") then

                    return svc.RF.Roll
                end
            end
        end
    end

    return nil
end

local rollRF = getRollRF()

local function checkRes(a, b)
    local ids = {}

    local function col(t)
        if type(t) ~= "table" then
            return
        end

        if t.Id or t.Style or t.Rarity then
            table.insert(ids, t.Id or t.Style)
        else
            for _, v in ipairs(t) do
                if type(v) == "string" then
                    table.insert(ids, v)
                end
            end
        end
    end

    col(b)
    col(a)

    local id = ids[#ids]

    if id and StyleMod then
        local ok, d = pcall(function()
            return StyleMod:Get(id)
        end)

        if ok and d then
            return d.Rarity, id
        end
    end

    return nil, id
end

local function doRoll(isLucky)
    if rollRF then
        local ok, a, b = pcall(
            rollRF.InvokeServer,
            rollRF,
            isLucky
        )

        return ok, a, b
    else
        local ok, a, b = pcall(function()
            local r1, r2 = StyleService:Roll(isLucky):expect()
            return r1, r2
        end)

        return ok, a, b
    end
end

-- ROLLBACK DENGAN PENGAMAN
local function doRollback()

    if targetFound or hasFirstTargetStyle() then
        stopSuccess()
        return
    end

    -- Tunggu sinkronisasi data style
    for i = 1, 5 do
        task.wait(0.3)

        if targetFound or hasFirstTargetStyle() then
            stopSuccess()
            return
        end
    end

    if targetFound or hasFirstTargetStyle() then
        stopSuccess()
        return
    end

    print("[AutoSpin] Timeskip Okazu tidak terdeteksi -> rollback")
    mamoriBudget.markRollback()

    local ok = pcall(function()
        local r = RS
            :WaitForChild("Packages")
            :WaitForChild("_Index")
            :WaitForChild("sleitnick_knit@1.7.0")
            :WaitForChild("knit")
            :WaitForChild("Services")
            :WaitForChild("SettingsService")
            :WaitForChild("RF")
            :WaitForChild("UpdateKeybind")

        r:InvokeServer(
            "MouseButton1",
            true,
            "Spike\xE2\x80\x8B\x8F"
        )

        TeleportService:TeleportToPlaceInstance(
            game.PlaceId,
            game.JobId
        )
    end)

    if not ok then
        Player:Kick("Rollback gagal")
    end
end

-- MAIN LOOP
task.spawn(function()

    while running do

        -- CEK TARGET SETIAP LOOP
        if hasFirstTargetStyle() then
            stopSuccess()
            break
        end

        local spins, lucky = nil, nil

        pcall(function()
            spins = StyleCtrl.Spins and StyleCtrl.Spins:get()
        end)

        pcall(function()
            lucky = StyleCtrl.LuckySpins and StyleCtrl.LuckySpins:get()
        end)
        -- Saldo nil berarti data belum siap, bukan Lucky 0.
        -- Jangan mengurangi budget berdasarkan snapshot sementara.
        if type(spins) ~= "number" or type(lucky) ~= "number" then
            task.wait(0.5)
            continue
        end
        observeFirstTargetLucky(lucky)

        if stopFirstTargetIfNoSafeSlot() then break end
        if mamoriBudget.capped() then
            print("[AutoSpin] BATAS " .. tostring(mamoriBudget.limit())
                .. " Lucky Timeskip Okazu tercapai -> rollback")
            saveFlag()
            doRollback()
            running = false
            break
        end
        if mamoriLuckyPendingBefore then
            local age = mamoriLuckyPendingSince and (os.clock() - mamoriLuckyPendingSince) or 0
            if os.clock() - lastFirstTargetPendingLog > 6 then
                lastFirstTargetPendingLog = os.clock()
                mamoriWatch("PENDING Lucky sebelum=" .. tostring(mamoriLuckyPendingBefore)
                    .. " sekarang=" .. tostring(lucky) .. " normal=" .. tostring(spins)
                    .. " tunggu=" .. tostring(math.floor(age)) .. "s; roll berikut ditahan")
            end
            -- Jangan lepaskan pending hanya karena hasil roll telah diterima.
            -- Baca ulang controller; bila masih sama, rejoin untuk mengambil
            -- snapshot saldo baru. Tidak pernah mengirim roll kedua ketika pending.
            if age >= 30 and os.clock() - mamoriRecoveryLastAttempt >= 20 then
                mamoriRecoveryLastAttempt = os.clock()
                mamoriRecoveryAttemptCount += 1
                local freshLucky
                local okFresh, freshErr = pcall(function()
                    local freshKnit = require(RS.Packages.Knit)
                    local freshCtrl = freshKnit.GetController("StyleController")
                    freshLucky = freshCtrl and freshCtrl.LuckySpins
                        and freshCtrl.LuckySpins:get() or nil
                end)
                mamoriWatch("RECOVERY #" .. tostring(mamoriRecoveryAttemptCount)
                    .. " | sebelum=" .. tostring(mamoriLuckyPendingBefore)
                    .. " | refresh=" .. tostring(freshLucky)
                    .. " | ok=" .. tostring(okFresh))
                if okFresh and type(freshLucky) == "number" then
                    observeFirstTargetLucky(freshLucky)
                end
                if mamoriLuckyPendingBefore then
                    mamoriWatch("RECOVERY REJOIN: saldo masih pending; tidak spin ulang")
                    local okTP, errTP = pcall(function()
                        TeleportService:Teleport(game.PlaceId, Player)
                    end)
                    if not okTP then
                        mamoriWatch("RECOVERY REJOIN gagal: " .. tostring(errTP)
                            .. "; pending dipertahankan")
                    else
                        -- Setelah permintaan rejoin diterima, hentikan worker
                        -- lokal; worker baru akan membaca saldo sesi baru.
                        mamoriWatch("RECOVERY REJOIN diminta; hentikan worker sesi lama")
                        running = false
                        break
                    end
                end
            end
            task.wait(0.5)
            continue
        end

        if spins <= 0 and lucky <= 0 then
            print("[AutoSpin] Semua spin habis -> STOP")
            running = false
            clearFlag()
            break
        end

        -- Satu pemeriksaan lokal terakhir sebelum menggunakan Lucky Style Spins.
        if ownedFromCache() then
            stopSuccess()
            break
        end
        mamoriWatch("ROLL REQUEST normal=" .. tostring(spins) .. " lucky=" .. tostring(lucky)
            .. " jenis=" .. (lucky > 0 and "LUCKY" or "NORMAL"))
        local ok, a, b = doRoll(lucky > 0)

        if not ok then
            warn("[AutoSpin] Roll ERROR: " .. tostring(a))
            task.wait(1)

        elseif a == false then
            print("[AutoSpin] Server menolak roll")
            task.wait(0.5)

        else
            local rar, id = checkRes(a, b)

            print("[AutoSpin] Rarity:", rar, "ID:", id)
            mamoriWatch("ROLL RESULT id=" .. tostring(id) .. " rarity=" .. tostring(rar))

            -- PRIORITAS UTAMA: TIMESKIP OKAZU
            local resultIsFirstTarget = false
            if id and StyleMod then
                local okDef, def = pcall(function() return StyleMod:Get(id) end)
                resultIsFirstTarget = okDef and def
                    and type(def.DisplayName) == "string"
                    and def.DisplayName:match("^%s*(.-)%s*$"):lower() == "timeskip okazu"
            end
            if resultIsFirstTarget then
                stopSuccess()
                break
            end

            -- CEK INVENTORY SETELAH SPIN
            if hasFirstTargetStyle() then
                stopSuccess()
                break
            end

            if lucky > 0 then
                mamoriLuckyPendingBefore = lucky
                mamoriLuckyPendingSince = os.clock()
                mamoriWatch("Lucky roll diterima; menunggu saldo turun dari " .. tostring(lucky)
                    .. " | hasil ID=" .. tostring(id) .. " rarity=" .. tostring(rar))
                -- Jangan mengirim roll Lucky berikutnya sebelum saldo
                -- mengonfirmasi pengurangan roll yang baru diterima.
                for _ = 1, 25 do
                    local afterValue
                    pcall(function() afterValue = StyleCtrl.LuckySpins:get() end)
                    observeFirstTargetLucky(afterValue)
                    if not mamoriLuckyPendingBefore then break end
                    if hasFirstTargetStyle() then
                        stopSuccess()
                        return
                    end
                    task.wait(0.2)
                end
                if mamoriLuckyPendingBefore then
                    warn("[AutoSpin] Roll Lucky Timeskip Okazu diterima tetapi saldo belum"
                        .. " turun; menunggu sinkronisasi, tidak spin lagi")
                    task.wait(0.5)
                    continue
                end
            end

            local luckyAfter = 0

            pcall(function()
                luckyAfter = StyleCtrl.LuckySpins
                    and StyleCtrl.LuckySpins:get()
                    or 0
            end)
            observeFirstTargetLucky(luckyAfter)

            if luckyAfter <= 0 or mamoriBudget.capped() then

                -- CEK SEBELUM ROLLBACK
                if hasFirstTargetStyle() then
                    stopSuccess()
                    break
                end

                print(mamoriBudget.capped()
                    and ("[AutoSpin] Batas " .. tostring(mamoriBudget.limit())
                        .. " Lucky tercapai, memeriksa Timeskip Okazu")
                    or "[AutoSpin] Lucky habis, memeriksa Timeskip Okazu")

                -- TUNGGU UPDATE DATA
                task.wait(1)

                if hasFirstTargetStyle() then
                    stopSuccess()
                    break
                end

                -- TIMESKIP OKAZU BELUM TERDETEKSI
                if stopFirstTargetIfNoSafeSlot() then break end
                saveFlag()
                doRollback()
                running = false
                break
            end
        end

        task.wait(0.55)
    end
end)


-- =====================================================================
-- TAMBAHAN TERPISAH: PEMANTAU ROLLBACK
-- Tidak mengedit doRollback(), doRoll(), atau MAIN LOOP asli.
-- =====================================================================
do
    local Log = game:GetService("LogService")
    local teleportCalled = false
    local rollbackMentioned = false
    local hasRolledThisSession = false
    local fallbackUsed = false
    local teleportRetries = 0
    local teleportStarted = false
    local rollErrors = 0

    -- Event ini hanya mengamati keluaran skrip inti. Jika log tidak keluar
    -- melalui LogService pada executor tertentu, fallback log tidak aktif.
    Log.MessageOut:Connect(function(message)
        if type(message) ~= "string" then return end
        if message:find("[AutoSpin] Timeskip Okazu SUDAH DIDAPATKAN", 1, true) then
            fallbackUsed = true
            return
        end
        if message:find("[AutoSpin] Rarity:", 1, true) then
            hasRolledThisSession = true
            rollErrors = 0
        end
        if message:find("[AutoSpin] Timeskip Okazu tidak terdeteksi -> rollback", 1, true) then
            rollbackMentioned = true
            teleportCalled = true
            task.delay(12, function()
                if not teleportStarted and not hasFirstTargetStyle() then
                    warn("[ROLLBACK GUARD] Rollback diminta, tetapi teleport belum terkonfirmasi. Periksa UpdateKeybind/TeleportService.")
                end
            end)
        end
        if message:find("[AutoSpin] Roll ERROR:", 1, true)
            or message:find("[AutoSpin] Server menolak roll", 1, true) then
            rollErrors = rollErrors + 1
            if rollErrors == 8 then
                warn("[ROLLBACK GUARD] Roll gagal 8 kali berturut-turut; skrip asli belum mencapai cabang rollback")
            end
        end
        if message:find("[AutoSpin] Semua spin habis -> STOP", 1, true)
            and hasRolledThisSession and not fallbackUsed and not rollbackMentioned then
            -- Skrip asli menghentikan diri pada kondisi semua spin 0.
            -- Fallback hanya kalau setidaknya ada roll berhasil sebelumnya.
            fallbackUsed = true
            task.spawn(function()
                task.wait(0.6)
                if hasFirstTargetStyle() then return end
                local stable = true
                for _ = 1, 4 do
                    local ok, spins, lucky = pcall(function()
                        return StyleCtrl.Spins:get(), StyleCtrl.LuckySpins:get()
                    end)
                    if not ok or type(spins) ~= "number" or type(lucky) ~= "number"
                        or spins > 0 or lucky > 0 then
                        stable = false
                        break
                    end
                    task.wait(0.4)
                end
                if not stable or hasFirstTargetStyle() or rollbackMentioned then
                    warn("[ROLLBACK GUARD] Stop 0-spin terjadi, tetapi data tidak stabil/target sudah ada; fallback dibatalkan")
                    return
                end
                warn("[ROLLBACK GUARD] Semua spin habis setelah roll; mencoba fungsi rollback ASLI sekali")
                saveFlag()
                local ok, err = pcall(doRollback)
                if not ok then
                    warn("[ROLLBACK GUARD] doRollback asli error:", tostring(err))
                end
            end)
        end
    end)

    -- Mengulang permintaan teleport hanya jika Roblox melaporkan gagal,
    -- dan hanya setelah cabang rollback asli benar-benar dipanggil.
    pcall(function()
        Player.OnTeleport:Connect(function(state)
            if state == Enum.TeleportState.Started or state == Enum.TeleportState.InProgress then
                teleportStarted = true
            end
        end)
    end)
    TeleportService.TeleportInitFailed:Connect(function(player, result, errorMessage)
        if player ~= Player or not rollbackMentioned or hasFirstTargetStyle() then return end
        warn("[ROLLBACK GUARD] Teleport gagal:", tostring(result), tostring(errorMessage))
        if teleportRetries >= 2 then
            warn("[ROLLBACK GUARD] Batas 2 percobaan ulang teleport tercapai")
            return
        end
        teleportRetries = teleportRetries + 1
        task.delay(2, function()
            if hasFirstTargetStyle() then return end
            local ok, err = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId)
            end)
            if not ok then
                warn("[ROLLBACK GUARD] Retry teleport error:", tostring(err))
            end
        end)
    end)
end

    end
    -- PENGAMAN PASCA-REJOIN (FIX): tunggu data StyleController TERBACA,
    -- bukan menunggu jumlah spin harus lebih dari nol.
    -- Nilai 0 yang valid harus diteruskan ke MAIN LOOP asli,
    -- yang menentukan sendiri apakah spin habis.
    local function waitForSpinData()
        __mamoriTrace("waitForSpinData() DIMULAI")
        local deadline = os.clock() + 30
        local lastStatus = ""
        local lastSpins, lastLucky = nil, nil
        local stableCount = 0
        while os.clock() < deadline do
            local ok, spins, lucky = pcall(function()
                local packages = RS:FindFirstChild("Packages")
                local knitModule = packages and packages:FindFirstChild("Knit")
                if not knitModule then return nil, nil end
                local knit = require(knitModule)
                local ctrl = knit.GetController("StyleController")
                if not ctrl then return nil, nil end
                local s = ctrl.Spins and ctrl.Spins:get()
                local l = ctrl.LuckySpins and ctrl.LuckySpins:get()
                return s, l
            end)
            if ok and type(spins) == "number" and type(lucky) == "number"
                and spins >= 0 and lucky >= 0 then
                if spins == lastSpins and lucky == lastLucky then
                    stableCount = stableCount + 1
                else
                    stableCount = 1
                end
                lastSpins, lastLucky = spins, lucky
                if stableCount >= 2 then
                    print("[FINAL] Data spin terbaca:", spins, "Lucky:", lucky)
                    __mamoriTrace("SPIN DATA SIAP spins="..tostring(spins).." lucky="..tostring(lucky))
                    return true
                end
                lastStatus = "menunggu angka stabil"
            else
                stableCount = 0
                lastStatus = "StyleController/data belum siap"
            end
            task.wait(0.5)
        end
        warn("[FINAL] Tidak dapat membaca data spin dalam 30 detik: " .. lastStatus)
        if typeof(writefile) == "function" then
            pcall(function()
                writefile("VL_TimeskipOkazu_Startup_Status.txt", "FAIL: " .. lastStatus)
            end)
        end
        return false
    end

    if waitForSpinData() then
        if typeof(writefile) == "function" then
            pcall(function()
                writefile("VL_TimeskipOkazu_Startup_Status.txt", "READY: Memulai AutoSpin. Jika spin dan lucky 0, MAIN LOOP asli akan STOP.")
            end)
        end
        __mamoriTrace("MEMANGGIL runTimeskipOkazu() SEKARANG")
        runTimeskipOkazu()
        __mamoriTrace("runTimeskipOkazu() KEMBALI")
    else
        __mamoriTrace("GAGAL waitForSpinData() -> AutoSpin tidak dimulai")
        -- Izinkan percobaan ulang pada sesi yang sama setelah gagal bootstrap.
        if mamoriSessionEnv[mamoriSessionKey] == game.JobId then
            mamoriSessionEnv[mamoriSessionKey] = nil
        end
    end
end
