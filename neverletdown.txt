-- v1.0.54: budget Lucky Spins per akun dan per fase, persisten saat rejoin.
-- Tidak mengubah fungsi spin, Pity, atau mekanisme rollback yang sudah ada.
local function makePersistentLuckyBudget(userId, phase, maximum)
    local TP = game:GetService("TeleportService")
    local env = (typeof(getgenv) == "function" and getgenv()) or _G
    local key = "VL_Lucky30_Budget_v154_" .. phase .. "_" .. tostring(userId)
    local filename = key .. ".txt"
    local state = {spent = 0, last = nil, rollbackPending = false}

    local function encode()
        return "v1|" .. tostring(state.spent) .. "|"
            .. tostring(state.last or -1) .. "|"
            .. (state.rollbackPending and "1" or "0")
    end
    local function decode(value)
        if type(value) ~= "string" then return false end
        local used, last, pending = value:match("^v1|(%d+)|([%-]?%d+)|([01])$")
        used, last = tonumber(used), tonumber(last)
        if not used or used < 0 or used > 100000
            or not last or last < -1 or last > 1000000 then return false end
        state.spent = used
        state.last = last == -1 and nil or last
        state.rollbackPending = pending == "1"
        return true
    end
    local loaded = false
    if typeof(readfile) == "function" then
        local ok, data = pcall(function()
            if typeof(isfile) == "function" and not isfile(filename) then return nil end
            return readfile(filename)
        end)
        if ok then loaded = decode(data) end
    end
    if not loaded then
        local ok, data = pcall(function() return TP:GetTeleportSetting(key) end)
        if ok then loaded = decode(data) end
    end
    if not loaded then loaded = decode(env[key]) end

    local function save()
        local encoded = encode()
        env[key] = encoded
        local fileOK = false
        if typeof(writefile) == "function" then
            fileOK = pcall(function() writefile(filename, encoded) end)
        end
        local tpOK = pcall(function() TP:SetTeleportSetting(key, encoded) end)
        if not fileOK and not tpOK then
            warn("[LUCKY BUDGET " .. phase .. "] Tidak bisa menyimpan budget "
                .. "lintas rejoin; periksa writefile/TeleportSetting")
        end
    end
    local function observe(balance)
        if type(balance) ~= "number" or balance < 0 then
            return state.spent, nil
        end
        local event = nil
        if state.last == nil then
            state.last = balance
            save()
            event = "INIT"
        elseif balance ~= state.last then
            if state.rollbackPending and balance > state.last then
                -- Rollback berhasil: saldo Lucky pulih. Mulai siklus 30 baru.
                state.spent = 0
                state.rollbackPending = false
                event = "RESET_AFTER_ROLLBACK"
            elseif balance < state.last then
                state.spent = state.spent + (state.last - balance)
                event = "SPENT"
            else
                -- Tambahan saldo tanpa rollback bukan alasan menghapus riwayat.
                event = "BALANCE_INCREASE"
            end
            state.last = balance
            save()
        end
        return state.spent, event
    end
    local function markRollback()
        state.rollbackPending = true
        save()
    end
    local function resetAfterSuccess()
        state.spent = 0
        state.last = nil
        state.rollbackPending = false
        save()
    end
    return {
        observe = observe,
        spent = function() return state.spent end,
        capped = function() return state.spent >= maximum end,
        markRollback = markRollback,
        resetAfterSuccess = resetAfterSuccess,
        isRollbackPending = function() return state.rollbackPending end,
        maximum = maximum,
        loaded = loaded
    }
end

                    local found = findDuplicateMamori()

                    end
                    if #found <= 1 then return true end
                    -- v1.0.47: Roll RF wajib ada untuk membersihkan duplikat.

                    if kijoDedicatedSlot then
                        for _, slot in ipairs(found) do
                            if slot == kijoDedicatedSlot then
                                extra = slot
                                break
                            end
                        end
                    end
                    local keep = nil
                    for _, slot in ipairs(found) do
                        if slot ~= extra then keep = slot; break end
                    end
                    if not keep then return false, "Tidak bisa menentukan Mamori yang dipertahankan" end
                    local lucky = readSignal(ctrl and ctrl.LuckySpins)
                    local normal = readSignal(ctrl and ctrl.Spins)
                    observeKijoLucky(lucky)
                    if type(lucky) ~= "number" or type(normal) ~= "number" then
                        return false, "Jumlah spin belum tersinkron"
                    end
                    if lucky <= 0 and normal <= 0 then
                        kijoStatus("Mamori ganda di slot " .. tostring(found[1])
                            .. " & " .. tostring(extra) .. "; spin habis, tidak bisa bersihkan")
                        return "NO_SPINS"
                    end
                    local useLucky = normal <= 0
                    if useLucky and kijoLuckyCapReached() then
                        return false, "LUCKY_CAP"
                    end
                    if useLucky and kijoLuckyPendingBefore then
                        return false, "LUCKY_PENDING"
                    end
                    kijoStatus("Mamori duplikat: pertahankan slot " .. tostring(keep)
                        .. ", spin ULANG slot " .. tostring(extra)
                        .. (useLucky and " (Lucky)" or " (Normal)"))

                    for _, slot in ipairs(before or {}) do
                        if slot == keep then stillKeep = true end
                        if slot == extra then stillExtra = true end
                    end

                        or not stillExtra

                        kijoStatus("Spin duplikat ditolak: " .. tostring(result))
                        return false, "Server menolak roll duplikat"

                            changed = true
                            break
                        end
                    end
                    if not changed then
                        kijoStatus("Spin duplikat belum terkonfirmasi; tahan spin berikutnya")
                        return false, "Hasil roll duplikat belum sinkron"
                    end
                    if useLucky then observeKijoLucky(readSignal(ctrl.LuckySpins)) end

                        .. tostring(after and #after or "belum sinkron"))

                local function initialSlotRank(id)
                    if type(id) ~= "string" or id == "" then
                        return nil, "slot belum siap"
                    end
                    local def = styleDef(id)
                    local dn = def and lower(def.DisplayName) or ""
                    local base = lower(id)
                    -- Target Mamori/Kijo mutlak dilindungi.
                    if base:find("mamori", 1, true) or base:find("kijo", 1, true)
                        or dn == "mamori" or dn == "kijo" then
                        return nil, "target dilindungi"
                    end
                    -- Dicek SEBELUM knownProtected (keduanya ada di sana).
                    if exceptionNames[base] or exceptionNames[dn] then
                        return 6, "fallback terakhir (Rintaro/Kimiro)"
                    end
                    if commonNames[base] or commonNames[dn] then return 1, "Common" end
                    if knownProtected[base] or knownProtected[dn] then
                        return nil, "Secret/Ultra/Evo dilindungi"
                    end
                    if not def then return nil, "data rarity tidak tersedia" end
                    local rarity = def.Rarity
                    if type(rarity) == "string" then
                        local r = lower(rarity)
                        if r == "secret" or r == "ultra" or r == "evo" or r == "evolution" then
                            return nil, "rarity " .. r .. " dilindungi"
                        end
                        if r == "common" or r == "uncommon" then return 1, r end
                        if r == "rare" then return 2, r end
                        if r == "legendary" then return 3, r end
                        if r == "godly" then return 4, r end
                        return nil, "rarity string tidak dikenal: " .. r
                    end
                    if type(rarity) == "number" then
                        if not calibrationComplete then calibrateProtectedRarity() end
                        if rarityProtectedNumbers[rarity] ~= nil then
                            return nil, "rarity numeric " .. tostring(rarity)
                                .. " dilindungi (" .. rarityProtectedNumbers[rarity] .. ")"
                        end
                        if calibrationComplete then
                            return 5, "non-Secret/Ultra/Evo (R=" .. tostring(rarity) .. ")"
                        end
                        return nil, "kalibrasi rarity belum lengkap"
                    end
                    return nil, "rarity belum terbaca"
                end
                local function slotSummary(slots)
                    if type(slots) ~= "table" then return "inventori belum tersedia" end
                    local parts = {}
                    for key, id in pairs(slots) do
                        local index = tonumber(key)
                        if index and index % 1 == 0 then
                            local _, label = initialSlotRank(id)
                            parts[#parts+1] = { index=index, text=tostring(index)
                                .. ":" .. tostring(id) .. " (" .. tostring(label) .. ")" }
                        end
                    end
                    table.sort(parts, function(x,y) return x.index < y.index end)
                    local result = {}
                    for i=1,math.min(#parts,8) do result[#result+1] = parts[i].text end
                    return table.concat(result, "; ")
                end

                -- v1.0.51: bedakan slot yang benar-benar penuh style terlindungi
                -- dari data slot/rarity yang belum tersinkron.
                -- true = ada slot aman; false = pasti tidak ada; nil = belum pasti.
                local function kijoSlotAvailability()
                    local slots = readSignal(teamA and teamA.Styles)
                    if type(slots) ~= "table" then
                        return nil, "data slot belum tersedia"
                    end
                    local knownSlots, uncertain = 0, 0
                    for key, id in pairs(slots) do
                        local idx = tonumber(key)
                        if idx and idx >= 1 and idx % 1 == 0 then
                            if type(id) ~= "string" or id == "" then
                                uncertain += 1
                            else
                                knownSlots += 1
                                local rank, reason = initialSlotRank(id)
                                if rank then
                                    return true, "slot " .. tostring(idx) .. " (" .. tostring(reason) .. ")"
                                end
                                -- Hanya 'dilindungi' yang membuktikan slot tidak aman.
                                -- Rarity belum terbaca TIDAK dianggap slot habis.
                                if type(reason) ~= "string"
                                    or not reason:find("dilindungi", 1, true) then
                                    uncertain += 1
                                end
                            end
                        end
                    end
                    if knownSlots == 0 or uncertain > 0 then
                        return nil, "slot/rarity belum pasti (terbaca="
                            .. tostring(knownSlots) .. ", belum siap=" .. tostring(uncertain) .. ")"
                    end
                    return false, "semua slot terlindungi: " .. slotSummary(slots)
                end

                -- v1.0.47: lebih tahan. SelectSlot ditunggu 10 detik; data yang
                -- belum sinkron dilaporkan, bukan membuat worker mati.
                local function chooseSafeStyleSlot(selectRF)
                    local slots = readSignal(teamA.Styles)
                    local active = tonumber(readSignal(teamA.CurrentStyleSlot))
                    if type(slots) ~= "table" or not active then
                        return false, "data slot/slot aktif belum siap"
                    end
                    local picked = kijoDedicatedSlot
                    if picked then
                        local id = slotValue(slots, picked)
                        if type(id) ~= "string" or id == "" then
                            return false, "slot khusus belum tersinkron"
                        end
                        if not initialSlotRank(id) then
                            print("[KIJO] Slot " .. tostring(picked) .. " kini dilindungi: " .. tostring(id))
                            picked, kijoDedicatedSlot = nil, nil
                        end
                    end
                    if not picked then
                        local bestRank = math.huge
                        for key, id in pairs(slots) do
                            local n = tonumber(key)
                            if n and n >= 1 and n % 1 == 0 then
                                local rank = initialSlotRank(id)
                                if rank and (rank < bestRank or (rank == bestRank and n == active)) then
                                    picked, bestRank = n, rank
                                end
                            end
                        end
                        if not picked then
                            return false, "tak ada slot aman: " .. slotSummary(slots)
                        end
                    end
                    if picked ~= active then
                        local ok, res = pcall(function() return selectRF:InvokeServer(picked) end)
                        if not ok or res == false then
                            return false, "SelectSlot gagal: " .. tostring(res)
                        end
                        local switched = false
                        for _ = 1, 50 do -- 10 detik
                            if tonumber(readSignal(teamA.CurrentStyleSlot)) == picked then
                                switched = true; break
                            end
                            task.wait(0.2)
                        end
                        if not switched then return false, "SelectSlot belum tersinkron" end
                    end
                    local id = slotValue(readSignal(teamA.Styles), picked)
                    if not initialSlotRank(id) then return false, "slot baru tidak aman" end
                    kijoDedicatedSlot = picked
                    return true
                end

                        if depth >= 2 then return "{...}" end
                        local parts, n = {}, 0
                        for k, v in pairs(x) do
                            n += 1
                            if n > 8 then parts[#parts+1] = "..."; break end
                            parts[#parts+1] = tostring(k) .. "=" .. describeResult(v, depth + 1)

                local function runKijoSpin()
                    if kijoStarted or kijoRollbackInFlight then return end
                    kijoStarted = true
                    kijoStatus("Memulai Kijo worker")
                    local ctrl, rollRF, selectRF
                    local lastContextStatus = -100
                    while playerA.Parent do
                        ctrl = getStyleContext()
                        rollRF = rollRF or findStyleRemote("Roll")
                        selectRF = selectRF or findStyleRemote("SelectSlot")
                        -- v1.0.52: bila semua slot Kijo terlindungi, final
                        -- webhook tidak perlu menunggu remote SelectSlot.
                        -- Bila masih ada slot aman, syarat remote tetap seperti semula.
                        if ctrl and selectRF then break end
                        if ctrl then
                            local available = kijoSlotAvailability()
                            if available == false then break end
                        end
                        if os.clock() - lastContextStatus > 5 then
                            lastContextStatus = os.clock()
                            kijoStatus("Menunggu konteks Style: controller=" .. tostring(ctrl ~= nil)
                                .. ", Roll=" .. tostring(rollRF ~= nil)
                                .. ", SelectSlot=" .. tostring(selectRF ~= nil))
                        end
                        task.wait(0.3)
                    end
                    if not playerA.Parent then return end
                    observeKijoLucky(readSignal(ctrl and ctrl.LuckySpins))
                    local kijoBefore = hasKijoStyle()
                    if kijoBefore == nil then
                        kijoStatus("Inventory Kijo belum siap; tunggu sinkronisasi")
                        return
                    end
                    -- Cek sebelum komit Lucky, pemilihan Pity, rejoin, atau rollback.
                    if kijoBefore == false then
                        local available, detail = kijoSlotAvailability()
                        if available == false then
                            if stopKijoNoSlot(detail) then return end
                            kijoStatus("Status slot berubah saat konfirmasi; periksa ulang")
                            return
                        elseif available == nil then
                            kijoStatus("Menunggu kepastian slot Kijo: " .. tostring(detail))
                            return
                        end
                    end
                    -- Periksa saldo Lucky SATU KALI saat fase Kijo dimulai
                    -- dalam game INI, sebelum Pity/duplikat/spin.
                    -- GO tidak dibawa lintas rejoin. Setelah GO, saldo 3/2/1
                    -- di sesi yang sama tetap boleh di-spin sampai 0 / batas 30.
                    local luckyAtStart = readSignal(ctrl.LuckySpins)
                    if kijoBefore == false then
                        if type(luckyAtStart) ~= "number" then
                            kijoStatus("Lucky awal Kijo belum terbaca; jangan spin dulu")
                            return
                        end
                        if not toZeroWasCommitted() then
                            if luckyAtStart <= 3 then
                                finishToZero()
                                kijoStatus("Lucky AWAL GAME INI " .. tostring(luckyAtStart)
                                    .. " (<=3): STOP tanpa spin, kirim webhook")
                                notifyFinal(false, "LOW_LUCKY_KIJO")
                                kijoFinished = true
                                return
                            end
                            if not commitToZero(luckyAtStart) then return end
                            kijoStatus("Lucky AWAL GAME INI " .. tostring(luckyAtStart)
                                .. " (>=4): lanjut sampai 0 / maksimal 30;"
                                .. " tidak diwariskan ke game berikutnya")
                        else
                            kijoStatus("Lanjut keputusan game INI: Lucky awal "
                                .. tostring(kijoInitialLucky) .. "; saldo kini "
                                .. tostring(luckyAtStart)
                                .. ". Tidak cek ambang 4 saat spin")
                        end
                    end
                    if kijoBefore == false and toZeroWasCommitted()
                        and not readyAfterKijoPityRejoin() then
                        kijoStatus("Menunggu pemilihan Secret Pity Kijo / rejoin")
                        return
                    end
                    local dupOK, dupWhy = clearAllDuplicates(ctrl, rollRF, selectRF)
                    observeKijoLucky(readSignal(ctrl.LuckySpins))
                    if dupOK == false then
                        if dupWhy == "LUCKY_CAP" or kijoLuckyCapReached() then
                            kijoRollbackOrNoSlot("Batas 30 Lucky Kijo saat pembersihan duplikat")
                            return
                        end
                        kijoStatus("Duplikat Mamori belum tuntas: " .. tostring(dupWhy))
                        return
                    end
                    if hasKijoStyle() == true then
                        kijoStatus("Kijo sudah dimiliki: STOP")
                        print("[KIJO] Kijo sudah dimiliki; tidak perlu spin.")
                        kijoDone()
                        return
                    end
                    local lucky = readSignal(ctrl.LuckySpins)
                    observeKijoLucky(lucky)
                    if type(lucky) ~= "number" then
                        kijoStatus("Lucky Style Spins belum dimuat")
                        warn("[KIJO] Lucky Style Spins belum dapat dibaca; tidak melakukan spin.")
                        return
                    end
                    -- Tidak ada pemeriksaan ulang ambang 4 di sini.
                    -- Komit GO sudah dibuat sebelum pembersihan duplikat.
                    if not readyAfterKijoPityRejoin() then
                        kijoStatus("Menunggu Kijo Pity / rejoin penyimpanan")
                        kijoStarted = false
                        return
                    end
                    if lucky <= 0 or kijoLuckyCapReached() then
                        kijoStatus(kijoLuckyCapReached()
                            and "Batas 30 Lucky Kijo tercapai: ROLLBACK"
                            or "Lucky 0, pencarian Kijo AKTIF: ROLLBACK seperti Mamori")
                        kijoRollbackOrNoSlot()
                        return
                    end
                    if kijoLuckyPendingBefore then
                        kijoStatus("Menunggu konfirmasi Lucky Kijo sebelumnya; tahan spin baru")
                        return
                    end

                    -- Pilih slot sekali (dengan ulang), seperti Mamori.
                    local slotReady, slotReason
                    for _ = 1, 5 do
                        slotReady, slotReason = chooseSafeStyleSlot(selectRF)
                        if slotReady then break end
                        local available, detail = kijoSlotAvailability()
                        if available == false then
                            if stopKijoNoSlot(detail) then return end
                            return -- data berubah; worker recovery akan mencoba lagi
                        end
                        kijoStatus("Slot belum siap, ulang: " .. tostring(slotReason))
                        task.wait(1)
                    end
                    if not slotReady then return end
                    kijoStatus("Slot khusus " .. tostring(kijoDedicatedSlot) .. ", Lucky " .. tostring(lucky))

local function loadStage()
    local raw
    if typeof(readfile) == "function" then
        pcall(function()
            if typeof(isfile) ~= "function" or isfile(FILE) then raw = readfile(FILE) end
        end)
    end
    if type(raw) ~= "string" or raw == "" then
        pcall(function() raw = TP:GetTeleportSetting(KEY) end)
    end
    if type(raw) ~= "string" then return "LEVEL", "" end
    local stage, origin = raw:match("^(%u+)|?(.*)$")
    if stage == "REDEEM" or stage == "REJOIN" or stage == "ROLL" then
        return stage, origin or ""
    end
    return "LEVEL", ""
end
local function saveStage(stage, origin)
    local raw = stage .. "|" .. tostring(origin or "")
    ENV.__VL_FINAL_STAGE = raw
    if typeof(writefile) == "function" then
        pcall(function() writefile(FILE, raw) end)
    end
    pcall(function() TP:SetTeleportSetting(KEY, raw) end)
end

-- REJOIN ADAPTASI DARI LEVELING: public-server listing + JobId lain.
-- Tujuan server hop adalah game.PlaceId saat ini, mengikuti skrip leveling asli.
local function requestRejoin()
    local guardKey = "__VL_MAMORI_REJOIN_GUARD_" .. tostring(Player.UserId)
    if ENV[guardKey] == tostring(game.JobId) then
        __mamoriTrace("REJOIN sedang diawasi di server ini")
        return true
    end
    ENV[guardKey] = tostring(game.JobId)
    task.spawn(function()
        local HttpService = game:GetService("HttpService")
        local currentJob = tostring(game.JobId)
        local begun, failed = false, false
        local failDescription = ""
        local onTP = Player.OnTeleport:Connect(function(state)
            __mamoriTrace("TELEPORT STATE=" .. tostring(state))
            if state == Enum.TeleportState.Started or state == Enum.TeleportState.InProgress then
                begun = true
            end
        end)
        local onFail = TP.TeleportInitFailed:Connect(function(p, result, msg, placeId)
            if p == Player then
                failed = true
                failDescription = tostring(result).." / "..tostring(msg)
                __mamoriTrace("TELEPORT GAGAL="..failDescription.." place="..tostring(placeId))
            end
        end)
        local function waitForTeleport(seconds)
            local deadline = os.clock() + seconds
            while os.clock() < deadline do
                if begun then return true end
                if failed then return false end
                task.wait(0.2)
            end
            return begun
        end
        local function getCandidates()
            local out, cursor, seen = {}, nil, {}
            for pageNum = 1, 4 do
                local url = "https://games.roblox.com/v1/games/" .. tostring(game.PlaceId)
                    .. "/servers/Public?sortOrder=Asc&limit=100"
                if cursor and cursor ~= "" then
                    url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
                end
                local ok, decoded = pcall(function()
                    return HttpService:JSONDecode(game:HttpGet(url))
                end)
                if not ok or type(decoded) ~= "table" then
                    __mamoriTrace("SERVER LIST gagal page="..pageNum.." : "..tostring(decoded):sub(1,100))
                    break
                end
                for _, server in ipairs(decoded.data or {}) do
                    if type(server.id) == "string" and server.id ~= currentJob
                        and not seen[server.id]
                        and (tonumber(server.playing) or math.huge) < (tonumber(server.maxPlayers) or 0) then
                        seen[server.id] = true
                        out[#out+1] = {id=server.id, count=tonumber(server.playing) or 99}
                    end
                end
                cursor = decoded.nextPageCursor
                if not cursor or #out >= 25 then break end
            end
            table.sort(out, function(a,b)
                if a.count == b.count then return a.id < b.id end
                return a.count < b.count
            end)
            return out
        end
        __mamoriTrace("REJOIN SAME-PLACE | source="..tostring(game.PlaceId).." goal="..tostring(game.PlaceId))
        local candidates = getCandidates()
        __mamoriTrace("SERVER PUBLIK PLACE INI ditemukan="..#candidates)
        local total = math.min(#candidates, 4)
        for i = 1, total do
            if begun then break end
            failed = false
            local target = candidates[i].id
            __mamoriTrace("SERVER HOP "..i.." -> "..target)
            local ok, err = pcall(function()
                TP:TeleportToPlaceInstance(game.PlaceId, target, Player)
            end)
            __mamoriTrace("HOP REQUEST="..tostring(ok)..(ok and "" or " err="..tostring(err)))
            if ok and waitForTeleport(9) then break end
            task.wait(1)
        end
        if not begun then
            failed = false
            __mamoriTrace("FALLBACK: Teleport ke game.PlaceId")
            local ok, err = pcall(function() TP:Teleport(game.PlaceId, Player) end)
            __mamoriTrace("FALLBACK REQUEST="..tostring(ok)..(ok and "" or " err="..tostring(err)))
            if ok then waitForTeleport(12) end
        end
        if not begun then
            __mamoriTrace("REJOIN TERTAHAN; menunggu pindah ke server lain")
            __mamoriTrace("KETERANGAN="..failDescription)
            __mamoriTrace("Kirim log teleport; tidak ada Kick")
        else
            __mamoriTrace("TELEPORT MULAI; menunggu sesi server baru")
        end
        pcall(function() onTP:Disconnect() end)
        pcall(function() onFail:Disconnect() end)
        ENV[guardKey] = nil
    end)
    return true
end

local function tutorialVisible(obj)
    if not obj or not obj:IsA("GuiObject") or not obj.Visible then
        return false
    end

    local node = obj.Parent
    while node and node ~= playerGui do
        if node:IsA("GuiObject") and not node.Visible then
            return false
        end
        if node:IsA("ScreenGui") and not node.Enabled then
            return false
        end
        node = node.Parent
    end

    return true
end

local function tutorialActivate(button)
    if not button then
        return false
    end

    local activated = false

    if typeof(getconnections) == "function" then
        local okConnections, connections = pcall(function()
            return getconnections(button.Activated)
        end)

        if okConnections and type(connections) == "table" then
            for _, connection in ipairs(connections) do
                local okFire = pcall(function()
                    if type(connection.Fire) == "function" then
                        connection:Fire()
                    elseif type(connection.Function) == "function" then
                        connection.Function()
                    end
                end)

                if okFire then
                    activated = true
                end
            end
        end
    end

    if not activated and typeof(firesignal) == "function" then
        activated = pcall(function()
            firesignal(button.Activated)
        end)
    end

    return activated
end

local function clickTutorialCenter()
    local camera = workspace.CurrentCamera
    if not camera then return false end
    local size = camera.ViewportSize
    if size.X <= 0 or size.Y <= 0 then return false end
    local vim = game:GetService("VirtualInputManager")
    return pcall(function()
        local x, y = size.X / 2, size.Y / 2
        vim:SendMouseButtonEvent(x, y, 0, true, game, 0)
        task.wait(0.10)
        vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)
end

    local pos, size = button.AbsolutePosition, button.AbsoluteSize

        vim:SendMouseButtonEvent(x, y, 0, true, game, 0)
        task.wait(0.10)
        vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)
end

local function tutorialPromptOptions()
    if not playerGui or not playerGui.Parent then
        playerGui = player:FindFirstChild("PlayerGui")
    end

    local interface = playerGui and playerGui:FindFirstChild("Interface")
    local persistent = interface and interface:FindFirstChild("Persistent")

    local bgFrame = prompt and prompt:FindFirstChild("BGFrame")
    local container = bgFrame and bgFrame:FindFirstChild("Container")

    return container and container:FindFirstChild("Option1") or nil,
        container and container:FindFirstChild("Option2") or nil,

local function tutorialModeDetected()
    local gamemode = string.lower(tostring(RS:GetAttribute("Gamemode") or ""))
    local gamemodeId = string.lower(tostring(RS:GetAttribute("GamemodeId") or ""))

    if gamemode == "tutorial" or gamemodeId == "tutorial" then
        return true
    end

    local gameGui = tutorialGameGui()
    if not gameGui then
        return false
    end

    local tutorialFrame = gameGui:FindFirstChild("Tutorial")
    local tutorialPrompt = gameGui:FindFirstChild("TutorialPrompt")

    return tutorialVisible(tutorialFrame)
        or tutorialVisible(tutorialPrompt)
        or tutorialMainSkip() ~= nil
end

local function tutorialStillActive()
    local gamemode = string.lower(tostring(RS:GetAttribute("Gamemode") or ""))
    local gamemodeId = string.lower(tostring(RS:GetAttribute("GamemodeId") or ""))

    return gamemode == "tutorial"
        or gamemodeId == "tutorial"
        or tutorialMainSkip() ~= nil
end

local function waitForStableTutorialSkip(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds
    local stableSince = nil
    local lastButton = nil

    while __xynIsCurrentRun() and os.clock() < deadline do
        local button = tutorialMainSkip()

        if button then
            if button ~= lastButton then
                lastButton = button
                stableSince = os.clock()
            elseif stableSince and os.clock() - stableSince >= 0.45 then
                return button
            end
        else
            lastButton = nil
            stableSince = nil
        end

        task.wait(0.10)
    end

    return nil
end

local function waitForTutorialPrompt(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds

    while __xynIsCurrentRun() and os.clock() < deadline do
        local option1, option2, prompt = tutorialPromptOptions()

        if prompt and tutorialVisible(prompt) then
            return option1, option2, prompt
        end

        task.wait(0.05)
    end

    return nil, nil, nil
end

local function waitForTutorialEnd(timeoutSeconds)
    local deadline = os.clock() + timeoutSeconds
    local clearedAt = nil

    while __xynIsCurrentRun() and os.clock() < deadline do
        if not tutorialStillActive() then
            clearedAt = clearedAt or os.clock()
            -- Jangan anggap sukses jika tombol/GUI hilang sesaat saat transisi.
            if os.clock() - clearedAt >= 0.7 then
                return true
            end
        else
            clearedAt = nil
        end
        task.wait(0.12)
    end

    return false
end

local function runAutoSkipTutorialGate()
    -- Klik tengah dahulu agar tutorial/Coach bisa menampilkan tombol skip.
    task.wait(0.5)
    clickTutorialCenter()

    local detectDeadline = os.clock() + 4
    while __xynIsCurrentRun() and os.clock() < detectDeadline do
        if tutorialModeDetected() then break end
        task.wait(0.12)
    end
    if not __xynIsCurrentRun() then return true, false end
    if not tutorialModeDetected() then return false, true end

    local attempts = 0
    while __xynIsCurrentRun() do
        if waitForTutorialEnd(1.0) then
            return true, true
        end

        attempts += 1
        -- Gagal? Klik tengah lagi, lalu coba Skip lagi, tanpa pindah server.
        clickTutorialCenter()
        task.wait(0.6)
        if waitForTutorialEnd(1.0) then return true, true end

        local mainButton = waitForStableTutorialSkip(4)
        if mainButton then
            pressTutorialButton(mainButton)
            local option1, option2, prompt = waitForTutorialPrompt(3)
            if prompt and tutorialVisible(prompt) then
                pressTutorialButton(option1)
                if waitForTutorialEnd(2) then return true, true end

                -- Coba tombol skip bawaan lagi lalu Option2 seperti patch asli.
                mainButton = waitForStableTutorialSkip(4)
                if mainButton then
                    pressTutorialButton(mainButton)
                    option1, option2, prompt = waitForTutorialPrompt(3)
                    if prompt and tutorialVisible(prompt) then

            end

    end

local BAD_SERVER_FILE = "xyn_bad_servers.json"
local BAD_SERVER_TELEPORT_KEY = "__xynBadServerHistory"
local MAX_BAD_SERVERS = 50
local SERVER_PAGE_LIMIT = 100
local SERVER_MAX_PAGES = 6
local SERVER_FETCH_RETRIES = 4

local badServerOrder = {}
local badServerSet = {}
local rejoinStarted = false

local function normalizeJobId(value)
    if type(value) ~= "string" then
        return nil
    end
    value = value:match("^%s*(.-)%s*$")
    if value == "" then
        return nil
    end
    return value
end

local function rememberBadServer(jobId)
    jobId = normalizeJobId(jobId)
    if not jobId then
        return false
    end

    if badServerSet[jobId] then
        -- Pindahkan ke posisi terbaru agar trimming menyimpan yang paling relevan.
        for i = #badServerOrder, 1, -1 do
            if badServerOrder[i] == jobId then
                table.remove(badServerOrder, i)
                break
            end
        end
    else
        badServerSet[jobId] = true
    end

    table.insert(badServerOrder, jobId)

    while #badServerOrder > MAX_BAD_SERVERS do
        local removed = table.remove(badServerOrder, 1)
        if removed then
            badServerSet[removed] = nil
        end
    end

    return true
end

local function mergeBadServerList(value)
    if type(value) ~= "table" then
        return
    end

    for _, jobId in ipairs(value) do
        rememberBadServer(jobId)
    end
end

local function decodeBadServerList(raw)
    if type(raw) ~= "string" or raw == "" then
        return nil
    end

    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)

    if ok and type(decoded) == "table" then
        return decoded
    end

    return nil
end

local function loadBadServers()
    -- 1) TeleportSetting: berguna saat pindah server dalam client yang sama.
    pcall(function()
        local raw = TeleportService:GetTeleportSetting(BAD_SERVER_TELEPORT_KEY)
        if type(raw) == "table" then
            mergeBadServerList(raw)
        elseif type(raw) == "string" then
            mergeBadServerList(decodeBadServerList(raw))
        end
    end)

    -- 2) File executor: fallback/persistence jika Delta mendukung readfile/writefile.
    if typeof(readfile) == "function" then
        local ok, raw = pcall(function()
            if typeof(isfile) == "function" and not isfile(BAD_SERVER_FILE) then
                return nil
            end
            return readfile(BAD_SERVER_FILE)
        end)

        if ok and type(raw) == "string" then
            mergeBadServerList(decodeBadServerList(raw))
        end
    end
end

local function persistBadServers()
    local encoded = nil
    pcall(function()
        encoded = HttpService:JSONEncode(badServerOrder)
    end)

    if type(encoded) ~= "string" then
        return false
    end

    pcall(function()
        TeleportService:SetTeleportSetting(BAD_SERVER_TELEPORT_KEY, encoded)
    end)

    if typeof(writefile) == "function" then
        pcall(function()
            writefile(BAD_SERVER_FILE, encoded)
        end)
    end

    return true
end

loadBadServers()

local function stableStringHash(value)
    value = tostring(value or "")
    local h = 17
    for i = 1, #value do
        h = (h * 131 + string.byte(value, i)) % 2147483647
    end
    return h
end

local function fetchPublicServerPage(cursor)
    local url = "https://games.roblox.com/v1/games/"
        .. tostring(game.PlaceId)
        .. "/servers/Public?sortOrder=Asc&limit="
        .. tostring(SERVER_PAGE_LIMIT)

    if cursor and cursor ~= "" then
        url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
    end

    local okHttp, body = pcall(function()
        return game:HttpGet(url)
    end)

    if not okHttp or type(body) ~= "string" or body == "" then
        return nil
    end

    local okDecode, decoded = pcall(function()
        return HttpService:JSONDecode(body)
    end)

    if not okDecode or type(decoded) ~= "table" then
        return nil
    end

    return decoded
end

local function collectServerCandidates()
    local candidates = {}
    local seen = {}
    local cursor = nil
    local thisJob = currentJobId()

    for _ = 1, SERVER_MAX_PAGES do
        if not detectorIsCurrent() then
            break
        end

        local page = nil
        for retry = 1, SERVER_FETCH_RETRIES do
            page = fetchPublicServerPage(cursor)
            if page then
                break
            end
            task.wait(0.35 * retry)
        end

        if not page then
            break
        end

        if type(page.data) == "table" then
            for _, server in ipairs(page.data) do
                local id = normalizeJobId(server.id)
                local playing = tonumber(server.playing) or math.huge
                local maxPlayers = tonumber(server.maxPlayers) or 0

                if id
                    and id ~= thisJob
                    and not badServerSet[id]
                    and not seen[id]
                    and maxPlayers > 0
                    and playing < maxPlayers
                then
                    seen[id] = true
                    table.insert(candidates, {
                        id = id,
                        playing = playing,
                        maxPlayers = maxPlayers,
                    })
                end
            end
        end

        cursor = page.nextPageCursor
        if type(cursor) ~= "string" or cursor == "" then
            break
        end

        -- Biasanya satu-dua halaman sudah cukup. Jangan spam API kalau kandidat banyak.
        if #candidates >= 40 then
            break
        end
    end

    return candidates
end

local function chooseServerCandidate(candidates)
    if type(candidates) ~= "table" or #candidates == 0 then
        return nil
    end

    -- Sort dulu supaya semua client punya basis yang sama.
    table.sort(candidates, function(a, b)
        if a.playing == b.playing then
            return a.id < b.id
        end
        return a.playing < b.playing
    end)

    -- Sebarkan akun script ke server berbeda. UserId membuat index berbeda
    -- sehingga beberapa account yang hop bersamaan tidak mudah berkumpul lagi.
    local salt = stableStringHash(currentJobId())
    local uid = tonumber(player.UserId) or 0
    local index = ((uid + salt) % #candidates) + 1
    return candidates[index]
end

local function tryTeleportToServer(jobId)
    jobId = normalizeJobId(jobId)
    if not jobId or jobId == currentJobId() or badServerSet[jobId] then
        return false
    end

    queueSelfForNextTeleport()
    persistBadServers()

    local ok = pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, jobId, player)
    end)

    return ok
end

local function serverHopAway(reason)
    -- Tidak ada teleport/server-hop saat tutorial masih aktif.
    if tutorialStillActive() then
        warn("[STARTUP FIX] Server hop ditahan karena tutorial aktif: " .. tostring(reason))
        return false
    end
    if rejoinStarted or not detectorIsCurrent() then
        return false
    end

    rejoinStarted = true

    local oldJob = currentJobId()
    if oldJob then
        rememberBadServer(oldJob)
    end
    persistBadServers()

    -- Coba beberapa kali mengambil server publik yang berbeda.
    for attempt = 1, 5 do
        if not detectorIsCurrent() then
            return false
        end

        local candidates = collectServerCandidates()
        local chosen = chooseServerCandidate(candidates)

        if chosen and tryTeleportToServer(chosen.id) then
            -- Jika request teleport gagal secara asynchronous dan kita masih berada
            -- di server ini beberapa detik kemudian, buka lock agar bisa retry.
            task.delay(10, function()
                if detectorIsCurrent()
                    and rejoinStarted
                    and currentJobId() == oldJob
                then
                    rejoinStarted = false
                    task.spawn(function()
                        serverHopAway(reason or "retry")
                    end)
                end
            end)
            return true
        end

        task.wait(0.6 * attempt)
    end

    -- Last resort: regular teleport. Server lama sudah berada di blacklist;
    -- kalau Roblox mengirim kita kembali ke JobId yang sama, startup detector
    -- akan langsung hop lagi.
    queueSelfForNextTeleport()
    persistBadServers()

    local ok = pcall(function()
        TeleportService:Teleport(game.PlaceId, player)
    end)

    if not ok then
        rejoinStarted = false
        return false
    end

    task.delay(10, function()
        if detectorIsCurrent()
            and rejoinStarted
            and currentJobId() == oldJob
        then
            rejoinStarted = false
            task.spawn(function()
                serverHopAway(reason or "retry")
            end)
        end
    end)

    return true
end

local function handleListedCollision()
    if not detectorIsCurrent() then return end
    if collisionHandled or collisionChecking or rejoinStarted then return end

    if #getListedSnapshot() < 3 then
        return
    end

    collisionChecking = true

    -- Dua konfirmasi cepat supaya player replication sesaat tidak memicu hop palsu.
    task.wait(0.25)
    if not detectorIsCurrent() or #getListedSnapshot() < 3 then
        collisionChecking = false
        return
    end

    task.wait(0.25)
    if not detectorIsCurrent() or #getListedSnapshot() < 3 then
        collisionChecking = false
        return
    end

    collisionHandled = true
    collisionChecking = false
    serverHopAway("3plus_listed")
end

local function getStartupAveragePingMs()
    local total = 0
    local samples = 0
    for _ = 1, 4 do
        if not detectorIsCurrent() then
            return nil
        end
        local pingMs = getNetworkPingMs()
        if pingMs then
            total += pingMs
            samples += 1
        end
        task.wait(0.25)
    end
    if samples == 0 then
        return nil
    end
    return total / samples
end

end

local function runStartupDetection()
    task.wait(1)
    if not detectorIsCurrent() then
        return
    end

    -- Kalau Roblox menaruh kita kembali ke server yang pernah ditandai buruk,
    -- jangan tunggu kondisi lain: langsung hop ke JobId berbeda.
    local jobId = currentJobId()
    if jobId and isBadServer(jobId) then
        serverHopAway("returned_to_bad_server")
        return
    end

    local pingMs = getStartupAveragePingMs()
    if pingMs and pingMs > PING_LIMIT_MS then
        handleHighPing()
        return
    end
    local startupDeadline = os.clock() + 30
    while detectorIsCurrent() and os.clock() < startupDeadline do
        if #getListedSnapshot() >= 3 then
            handleListedCollision()
            if collisionHandled or collisionChecking then
                return
            end
        end
        task.wait(0.25)
    end
end

local knownTeamObjects = {

local function rememberTeamObject(teamId, teamObject)
    if not teamObject or not teamObject:IsA("Team") then

local function parseTeamIdValue(value)
    if typeof(value) == "number" then
        local n = tonumber(value)
        if n == 1 or n == 2 then
            return n
        end
        return nil
    end
    if typeof(value) == "string" then
        local lower = string.lower(value)
        local direct = tonumber(lower)
        if direct == 1 or direct == 2 then
            return direct
        end
        local fromName = lower:match("team%s*(%d+)") or lower:match("team(%d+)")
        fromName = tonumber(fromName)
        if fromName == 1 or fromName == 2 then
            return fromName
        end
    end
    return nil
end
local function extractTeamIdFromRemoteResult(result)
    local direct = parseTeamIdValue(result)
    if direct then
        return direct
    end
    if typeof(result) == "table" then
        for _, value in ipairs({
            result.TeamId,
            result.TeamID,
            result.teamId,
            result.teamID,
            result.Team,
            result.team,
            result.Id,
            result.id,
        }) do
            local parsed = parseTeamIdValue(value)
            if parsed then
                return parsed
            end
        end
    end
    return nil
end
local function joinTeam(teamId)
    if not refreshGameRemotes() then
        return false, nil
    end
    lastRequestedTeamId = teamId
    lastRequestAt = os.clock()
    local beforeTeam = player.Team
    local requestResult = nil
    local ok = pcall(function()
        requestResult = RequestJoin:InvokeServer(teamId, 5)
        Serve:InvokeServer(Vector3.zero, 0)
    end)
    local afterTeam = player.Team
    if ok and afterTeam and afterTeam ~= beforeTeam then
        rememberTeamObject(teamId, afterTeam)
    end
    local resultTeamId = extractTeamIdFromRemoteResult(requestResult)
    if ok and resultTeamId == teamId and afterTeam then
        rememberTeamObject(teamId, afterTeam)
    end
    return ok, requestResult
end
local function normalizeTeamValue(value)
    if value == nil then return nil end
    return string.lower(tostring(value)):gsub("[^%w]", "")
end
local function valueMatchesTeam(value, teamId)
    local normalized = normalizeTeamValue(value)
    if not normalized then return false end
    return normalized == tostring(teamId)
        or normalized == ("team" .. tostring(teamId))
end
local function refreshKnownTeamObjectsFromService()
    local serviceTeams = TeamsService:GetTeams()
    for _, teamObject in ipairs(serviceTeams) do
        for teamId = 1, 2 do
            local attr = teamObject:GetAttribute("TeamId")
                or teamObject:GetAttribute("TeamID")
                or teamObject:GetAttribute("Id")
            if valueMatchesTeam(teamObject.Name, teamId)
                or valueMatchesTeam(attr, teamId)
            then
                rememberTeamObject(teamId, teamObject)
            end
        end
    end
    if #serviceTeams == 2 then
        if knownTeamObjects[1] and not knownTeamObjects[2] then
            for _, teamObject in ipairs(serviceTeams) do
                if teamObject ~= knownTeamObjects[1] then
                    rememberTeamObject(2, teamObject)
                end
            end
        elseif knownTeamObjects[2] and not knownTeamObjects[1] then
            for _, teamObject in ipairs(serviceTeams) do
                if teamObject ~= knownTeamObjects[2] then
                    rememberTeamObject(1, teamObject)
                end
            end
        end
    end
end
local function instanceTeamMatches(instance, teamId)
    if not instance then return false end
    if knownTeamObjects[teamId] and instance == knownTeamObjects[teamId] then
        return true
    end
    if valueMatchesTeam(instance.Name, teamId) then
        return true
    end
    local attrs = {
        instance:GetAttribute("TeamId"),
        instance:GetAttribute("TeamID"),
        instance:GetAttribute("teamId"),
        instance:GetAttribute("Team"),
        instance:GetAttribute("team"),
    }
    for _, value in ipairs(attrs) do
        if valueMatchesTeam(value, teamId) then
            return true
        end
    end
    return false
end
local function containerTeamMatches(container, teamId)
    if not container then return false end
    if instanceTeamMatches(container, teamId) then
        return true
    end
    for _, childName in ipairs({"Team", "TeamId", "TeamID", "team", "teamId"}) do
        local child = container:FindFirstChild(childName)
        if child and child:IsA("ValueBase") and valueMatchesTeam(child.Value, teamId) then
            return true
        end
    end
    return false
end
local function teamLooksConfirmedFor(targetPlayer, teamId)
    if not targetPlayer then return false end
    refreshKnownTeamObjectsFromService()
    local teamObject = targetPlayer.Team
    if teamObject and instanceTeamMatches(teamObject, teamId) then
        return true
    end
    if containerTeamMatches(targetPlayer, teamId) then
        return true
    end
    local targetCharacter = targetPlayer.Character
    if targetCharacter and containerTeamMatches(targetCharacter, teamId) then
        return true
    end
    return false
end
local function teamLooksConfirmed(teamId)

end

local TEAM_SANITY_INTERVAL = 2.00
local teamGeneration = 0
local teamCycleRunningGeneration = nil
local teamCycleStartedAt = 0  -- v1.0.49: watchdog generasi; tidak melewati konfirmasi team.
local correctionRunningGeneration = nil
local correctionStartedAt = 0  -- v1.0.49: deteksi koreksi team yang tidak selesai
local teamJoinFinished = false
local lockedTeamId = nil
local lockedRosterSignature = nil
local roundResetActive = false
local function generationIsCurrent(generation)
    return getgenv().main
        and detectorIsCurrent()
        and generation == teamGeneration
end
local function waitForStableRoster(generation)
    local lastSignature = nil
    local stableSince = 0
    while generationIsCurrent(generation) do
        local snapshot = getPresentConfiguredPlayerObjects()
        local signature = getPresentSignatureFrom(snapshot)
        if signature ~= lastSignature then
            lastSignature = signature
            stableSince = os.clock()
        elseif os.clock() - stableSince >= TEAM_ROSTER_STABLE_TIME then
            return snapshot, signature
        end
        task.wait(0.15)
    end
    return nil, nil
end
local function waitForStableLocalConfirmation(generation, targetTeamId)
    local stableSince = nil
    while generationIsCurrent(generation) do
        if teamLooksConfirmed(targetTeamId) then
            stableSince = stableSince or os.clock()
            if os.clock() - stableSince >= TEAM_CONFIRM_STABLE_TIME then
                return true
            end
        else
            stableSince = nil
            return false
        end
        task.wait(0.10)
    end
    return false
end
local function findSameTeamPartner()
    local myTeam = player.Team
    for _, partner in ipairs(getPresentConfiguredPlayerObjects()) do
        if partner ~= player then
            local partnerTeam = partner.Team
            if myTeam and partnerTeam and myTeam == partnerTeam then
                return partner
            end
            for teamId = 1, 2 do
                if teamLooksConfirmedFor(player, teamId)
                    and teamLooksConfirmedFor(partner, teamId)
                then
                    return partner
                end
            end
        end
    end
    return nil
end
local function getKnownTeamIdFor(targetPlayer)
    for teamId = 1, 2 do
        if teamLooksConfirmedFor(targetPlayer, teamId) then
            return teamId
        end
    end
    return nil
end
local function pairSeparatedState(snapshot)
    if #snapshot ~= 2 then
        return true
    end
    local partner = snapshot[1] == player and snapshot[2] or snapshot[1]
    if not partner or partner == player then
        return true
    end
    if player.Team and partner.Team then
        return player.Team ~= partner.Team
    end
    local myTeamId = getKnownTeamIdFor(player)
    local partnerTeamId = getKnownTeamIdFor(partner)
    if myTeamId and partnerTeamId then
        return myTeamId ~= partnerTeamId
    end
    return nil
end
local function waitForPairSeparation(generation, snapshot, targetTeamId)
    if #snapshot ~= 2 then
        return true
    end
    local separatedSince = nil
    while generationIsCurrent(generation) do
        if not teamLooksConfirmed(targetTeamId) then
            return false
        end
        local state = pairSeparatedState(snapshot)
        if state == true then
            separatedSince = separatedSince or os.clock()
            if os.clock() - separatedSince >= 0.60 then
                return true
            end
        else
            separatedSince = nil
        end
        task.wait(0.10)
    end
    return false
end
local function runTeamCycle(generation)
    local snapshot, signature = waitForStableRoster(generation)
    if not snapshot or not generationIsCurrent(generation) then
        return false
    end
    local targetTeamId, position = getTargetTeamFromSnapshot(player, snapshot)
    if not targetTeamId then
        return false
    end
    lockedTeamId = targetTeamId
    lockedRosterSignature = signature
    teamJoinFinished = false
    if position then
        task.wait(0.12 + ((position - 1) * 0.18))
    end
    while generationIsCurrent(generation) do
        if teamLooksConfirmed(targetTeamId) then
            if waitForStableLocalConfirmation(generation, targetTeamId) then
                if waitForPairSeparation(generation, snapshot, targetTeamId) then
                    teamJoinFinished = true
                    return true
                end
            end
        end
        joinTeam(targetTeamId)
        task.wait(TEAM_RETRY_DELAY)
    end
    return false
end
local function launchTeamCycle()
    if roundResetActive or not getgenv().main then
        return
    end
    local generation = teamGeneration
    if teamCycleRunningGeneration == generation then
        return
    end
    teamCycleRunningGeneration = generation
    teamCycleStartedAt = os.clock()
    task.spawn(function()
        local ok, err = pcall(runTeamCycle, generation)
        if not ok then
            warn("[LEVEL RECOVERY] Team cycle error: " .. tostring(err))
        end
        if teamCycleRunningGeneration == generation then
            teamCycleRunningGeneration = nil
        end
    end)
end
local function invalidateTeamAssignment(startImmediately)
    teamGeneration = teamGeneration + 1
    teamJoinFinished = false
    lockedTeamId = nil
    lockedRosterSignature = nil
    correctionRunningGeneration = nil
    correctionStartedAt = 0
    if startImmediately and not roundResetActive then
        launchTeamCycle()
    end
end
local function initializeJoins()
    if not isConfiguredUsername(player.Name) then
        return false
    end
    if teamGeneration == 0 then
        invalidateTeamAssignment(true)
    else
        launchTeamCycle()
    end
    while getgenv().main and detectorIsCurrent() do
        if teamJoinFinished and lockedTeamId then
            return true
        end
        if not roundResetActive and teamCycleRunningGeneration == nil then
            launchTeamCycle()
        end
        task.wait(0.15)
    end
    return false
end
local function correctLockedTeam()
    if roundResetActive then
        return
    end
    if not lockedTeamId then
        launchTeamCycle()
        return
    end
    local generation = teamGeneration
    if correctionRunningGeneration == generation then
        return
    end
    correctionRunningGeneration = generation
    correctionStartedAt = os.clock()
    teamJoinFinished = false
    local targetTeamId = lockedTeamId
    task.spawn(function()
        local ok, err = pcall(function()
            while generationIsCurrent(generation)
                and not roundResetActive
                and lockedTeamId == targetTeamId
            do
                if teamLooksConfirmed(targetTeamId) then
                    if waitForStableLocalConfirmation(generation, targetTeamId) then
                        teamJoinFinished = true
                        break
                    end
                end
                joinTeam(targetTeamId)
                task.wait(TEAM_RETRY_DELAY)
            end
        end)
        if not ok then
            warn("[LEVEL RECOVERY] Team correction error: " .. tostring(err))
        end
        if correctionRunningGeneration == generation then
            correctionRunningGeneration = nil
            correctionStartedAt = 0
        end
    end)
end
local function sanityCheckTeam()
    if roundResetActive or not getgenv().main then
        return
    end
    local currentSignature = getPresentSignature()
    if lockedRosterSignature and currentSignature ~= lockedRosterSignature then
        invalidateTeamAssignment(true)
        return
    end
    if not lockedTeamId then
        launchTeamCycle()
        return
    end
    if not teamLooksConfirmed(lockedTeamId) then
        correctLockedTeam()
        return
    end
    -- v1.0.49: game mengonfirmasi team namun gate internal belum TRUE.
    -- Jika tak ada worker aktif, ulang konfirmasi, jangan memalsukan gate.
    if not teamJoinFinished
        and teamCycleRunningGeneration == nil
        and correctionRunningGeneration == nil then
        launchTeamCycle()
        return
    end
    local sameTeamPartner = findSameTeamPartner()
    if sameTeamPartner then
        local snapshot = getPresentConfiguredPlayerObjects()
        local myTarget = getTargetTeamFromSnapshot(player, snapshot)
        if myTarget and myTarget ~= lockedTeamId then
            invalidateTeamAssignment(true)
            return
        end
    end
end
local roundResetStartedAt = 0
local function beginRoundTeamReset()
    if roundResetActive then return end
    roundResetActive = true
    roundResetStartedAt = os.clock()
    teamGeneration = teamGeneration + 1
    teamJoinFinished = false
    lockedTeamId = nil
    lockedRosterSignature = nil
    correctionRunningGeneration = nil
    correctionStartedAt = 0
end
local function finishRoundTeamReset()
    if not roundResetActive then return end
    roundResetActive = false
    roundResetStartedAt = 0
    invalidateTeamAssignment(true)
end

    end)

local function returnVisiblyOpen(obj)
    if not obj or not obj.Parent then return false end
    local node = obj
    while node and node ~= game do
        if node:IsA("GuiObject") and not node.Visible then return false end
        if node:IsA("ScreenGui") and not node.Enabled then return false end
        node = node.Parent
    end
    return true
end

    local button = container and container:FindFirstChild("BackBtn")

local function pressRoundReturn(button)
    if typeof(getconnections) == "function" then
        local ok, connections = pcall(function()
            return getconnections(button.Activated)
        end)
        if ok and type(connections) == "table" then
            for _, connection in ipairs(connections) do
                local fired = pcall(function()
                    if typeof(connection.Fire) == "function" then
                        connection:Fire()
                    elseif typeof(connection.Function) == "function" then
                        connection.Function()
                    else
                        error("Callback tidak tersedia")
                    end
                end)
                if fired then return true end
            end
        end
    end
    if typeof(firesignal) == "function" then
        local ok = pcall(function() firesignal(button.Activated) end)
        if ok then return true end
    end
    local ok = pcall(function()
        local x = button.AbsolutePosition.X + button.AbsoluteSize.X / 2
        local y = button.AbsolutePosition.Y + button.AbsoluteSize.Y / 2
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
        task.wait(0.08)
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)
    return ok
end

local startupGatePassed = false
local startupState = "WAIT_TUTORIAL"
local startupWorkerActive = false
local startupLastAttempt = -100
local function launchStartupWorker()
    if startupGatePassed or startupWorkerActive or not detectorIsCurrent() then return end
    if os.clock() - startupLastAttempt < 8 then return end
    startupLastAttempt = os.clock()
    startupWorkerActive = true
    task.spawn(function()
        local ok, err = pcall(function()
            startupState = "WAIT_TUTORIAL"
            while detectorIsCurrent() and not tutorialGateFinished do
                task.wait(0.20)
            end
            if not detectorIsCurrent() then return end
            if not tutorialGateCleared or tutorialStillActive() then
                startupState = "TUTORIAL_NOT_CLEARED"
                return
            end
            startupState = "WAIT_TEAM"
            warn("[LEVEL RECOVERY] Tutorial cleared -> starting team confirmation")
            local confirmed = initializeJoins()
            if not confirmed or not detectorIsCurrent() then return end
            startupState = "SERVER_CHECK"
            runStartupDetection()
            if detectorIsCurrent() and not rejoinStarted then
                startupGatePassed = true
                startupState = "READY"
                warn("[LEVEL RECOVERY] Startup gate READY")
            end
        end)
        if not ok then
            startupState = "ERROR: " .. tostring(err):sub(1,90)
            warn("[LEVEL RECOVERY] Startup worker error: " .. tostring(err))
        end
        startupWorkerActive = false
    end)
end

local function gateReason()
    if not tutorialGateFinished or not tutorialGateCleared then return "TUTORIAL" end
    if roundResetActive then return "RETURN_RESET" end
    if not startupGatePassed then return "STARTUP:" .. startupState end
    if not lockedTeamId then return "NO_LOCKED_TEAM" end
    if not teamJoinFinished then return "TEAM_UNCONFIRMED" end
    return "READY"
end

task.spawn(function()
    local lastReport = ""
    while detectorIsCurrent() do
        task.wait(1)
        local now = os.clock()
        local blocked = gateReason()
        if blocked == "READY" then
            gateBlockedSince = nil
        else
            gateBlockedSince = gateBlockedSince or now
        end
        local blockedFor = gateBlockedSince and (now - gateBlockedSince) or 0

        -- Backup saat thread Return berhenti/tidak menutup reset state.
        -- Jangan reset saat tombol Return MASIH terlihat: urutan Return tetap wajib.
        if roundResetActive and now - roundResetStartedAt >= 4 then
            local visible = getRoundReturnButton()
            if not visible then
                recoveryNote("RETURN hidden -> restore team assignment")
                finishRoundTeamReset()
                returnClicker = false
            elseif now - returnLastAttempt >= 5 then
                returnLastAttempt = now
                pcall(pressRoundReturn, visible)
                recoveryNote("RETURN visible -> retry click")
            end
        end

        -- Jika team cycle lokal macet >45 detik, buat generasi baru.
        -- Tetap lewat pemeriksaan roster, team, dan partner seperti mekanisme asli.
        if not roundResetActive and tutorialGateCleared and blockedFor >= 18
            and now - lastTeamRecovery >= 15 and getgenv().main then
            lastTeamRecovery = now
            if not startupGatePassed and not startupWorkerActive then
                recoveryNote("STARTUP worker ended -> retry")
                launchStartupWorker()
            elseif startupState == "WAIT_TEAM" or startupGatePassed then
                if teamCycleRunningGeneration ~= nil
                    and now - teamCycleStartedAt >= 45 then
                    recoveryNote("TEAM cycle stalled >45s -> re-evaluate roster")
                    invalidateTeamAssignment(true)
                elseif correctionRunningGeneration ~= nil
                    and now - correctionStartedAt >= 45 then
                    recoveryNote("TEAM correction stalled >45s -> re-evaluate roster")
                    invalidateTeamAssignment(true)
                elseif not teamJoinFinished then
                    if teamCycleRunningGeneration == nil and correctionRunningGeneration == nil then
                        recoveryNote("TEAM not confirmed -> retry team cycle")
                        if lockedTeamId and not teamLooksConfirmed(lockedTeamId) then
                            correctLockedTeam()
                        else
                            launchTeamCycle()
                        end
                    else
                        recoveryNote("TEAM confirmation in progress")
                    end
                end
            end
        end

        -- Worker aksi dapat mati secara diam-diam walau boot tetap true.
        -- Restart hanya bila gate benar-benar READY; generasi lama akan keluar
        -- pada iterasi berikutnya, bukan dua loop aktif permanen.
        local envRecovery = getgenv()
        local workerTick = tonumber(envRecovery.__xynLevelActionTick149)
        local workerAge = workerTick and (now - workerTick) or 0
        if gateReason() == "READY" and workerTick and workerAge > 20
            and now - lastGameplayWorkerRecovery >= 40 then
            lastGameplayWorkerRecovery = now
            if typeof(envRecovery.__xynRestartLevelActions149) == "function" then
                recoveryNote("GAMEPLAY worker unresponsive >20s -> restart worker")
                pcall(envRecovery.__xynRestartLevelActions149)
            end
        end

        ensureRecoveryPanel()
        local report = table.concat({
            "LEVEL RECOVERY v1.0.49 [TAP COPY]",
            "User=" .. player.Name .. " | Lv=" .. tostring(player:GetAttribute("User_Level")),
            "Gate=" .. gateReason() .. " | Stuck=" .. tostring(math.floor(blockedFor)) .. "s",
            "Tutorial=" .. tostring(tutorialGateCleared) .. " | Startup=" .. tostring(startupGatePassed),
            "Team=" .. tostring(player.Team and player.Team.Name or "nil") .. " | ID=" .. tostring(lockedTeamId),
            "TeamOK=" .. tostring(teamJoinFinished) .. " | Reset=" .. tostring(roundResetActive),
            "TeamCycle=" .. tostring(teamCycleRunningGeneration) .. " | Corr=" .. tostring(correctionRunningGeneration),
            "WorkerAge=" .. tostring(math.floor(workerAge)) .. "s",
            "Note=" .. recoveryLastReason,
        }, "\n")
        recoveryText = report
        local button = recoveryPanel and recoveryPanel:FindFirstChild("CopyLevelRecovery")
        if button then button.Text = report end
        if gateReason() ~= lastReport then
            lastReport = gateReason()
            warn("[LEVEL RECOVERY] Gate=" .. lastReport)
        end
    end
end)

                warn("[STYLE] Tidak dapat membaca slot; AutoSpin ditahan.")
                return false
            end

            -- PRIORITAS TARGET: bila target sudah ada di inventory,
            -- jangan pernah menjalankan seleksi slot atau Kick karena slot aman habis.
            -- Deteksi Mamori dari DisplayName setiap slot, bukan ID internal.
            for _, entry in ipairs(all) do
                if isTargetDisplay(entry.id) then
                    print("[STYLE] Target sudah dimiliki (" .. tostring(entry.id) .. "); lewati pemeriksaan slot aman.")
                    return true
                end
            end

            local commonTarget, normalTarget, exceptionTarget, activeEntry
            for _, s in ipairs(all) do
                if s.slot == active then activeEntry = s end
                if s.cls == "COMMON" and (not commonTarget or s.slot == active) then
                    commonTarget = s
                elseif s.cls == "AMAN" and (not normalTarget or s.slot == active) then
                    normalTarget = s
                elseif s.cls == "PENGECUALIAN" and (not exceptionTarget or s.slot == active) then
                    exceptionTarget = s
                end
            end
            local target = commonTarget or normalTarget or exceptionTarget
            if not target then
                -- v1.0.52: hanya final jika SEMUA slot pasti terlindungi.
                -- Data yang belum siap tidak boleh memicu webhook final.
                local function protectedSnapshot()
                    local raw = getStyles()
                    local entries = scan()
                    if type(raw) ~= "table" or type(entries) ~= "table"
                        or #entries == 0 then return nil end
                    local numericCount = 0
                    for key in pairs(raw) do
                        local n = tonumber(key)
                        if n and n >= 1 and n % 1 == 0 then
                            numericCount += 1
                        end
                    end
                    if numericCount ~= #entries then return nil end
                    local signature = {}
                    for _, entry in ipairs(entries) do
                        if entry.cls ~= "TERLINDUNGI" then return nil end
                        signature[#signature+1] = tostring(entry.slot)
                            .. "=" .. tostring(entry.id)
                    end
                    return table.concat(signature, "|")
                end
                local first = protectedSnapshot()
                if first then
                    task.wait(1)
                    local second = protectedSnapshot()
                    if second and first == second then
                        local env = (typeof(getgenv) == "function" and getgenv()) or _G
                        local uid = tostring(player.UserId)
                        env["__VL_MAMORI_NO_SLOT_" .. uid] = true
                        print("[STYLE] STOP Mamori: semua slot terlindungi;"
                            .. " kirim webhook final berisi semua style langka.")
                        local notifier = env["__VL_FINAL_NOTIFY_" .. uid]
                        if type(notifier) == "function" then
                            pcall(notifier, false, "NO_SLOT_MAMORI")
                        end
                        return false
                    end
                end
                warn("[STYLE] Belum ada slot aman, tetapi data slot/rarity"
                    .. " belum pasti. AutoSpin ditahan; tanpa Kick/webhook palsu.")
                return false
            end

            if not selectSlotRF then
                warn("[STYLE] SelectSlot belum tersedia; AutoSpin ditahan.")
                return false
            end

            if activeEntry and activeEntry.slot == target.slot then
                return true -- Sudah aktif; lanjut ke skrip berikutnya tanpa rejoin.
            end

            local ok, response = pcall(function()
                return selectSlotRF:InvokeServer(target.slot)
            end)
            if not ok then
                warn("[STYLE] SelectSlot gagal: " .. tostring(response))
                return false
            end
            local verifyDeadline = os.clock() + 6
            while os.clock() < verifyDeadline do
                local now = getCurrent()
                if now == target.slot then
                    local styles = getStyles()
                    if styles and styles[now] == target.id then
                        rejoinPending = true
                        task.wait(0.6)
                        if player and player.Parent then
                            local sent, err = pcall(function()
                                TeleportService:Teleport(game.PlaceId, player)
                            end)
                            if not sent then
                                rejoinPending = false
                                warn("[STYLE] Permintaan rejoin gagal: " .. tostring(err))
                            end
                        end
                        -- Jangan jalankan AutoSpin di server lama setelah mengganti slot.
                        return false
                    end
                end
                task.wait(0.2)
            end
            warn("[STYLE] Pergantian slot belum terverifikasi; AutoSpin ditahan.")
            return false
        end

        -- Jika slot baru dipilih, rejoin sedang dicoba lalu skrip ini berhenti;
        -- jika semua slot terlindungi, webhook final. Jika slot benar, lanjut.
        __mamoriTrace("SEBELUM selectSafeStyle()")
        if not selectSafeStyle() then
            __mamoriTrace("selectSafeStyle() FALSE -> keluar sebelum AutoSpin")
            return
        end
        __mamoriTrace("selectSafeStyle() TRUE -> lanjut")
    end
end

-- CACHE KEPEMILIKAN MAMORI PER AKUN (dipercaya tanpa menunggu Style dimuat)
local OWNED_FILE = "VL_Mamori_Owned_" .. tostring(Player.UserId) .. ".flag"
local OWNED_KEY = "__VL_MAMORI_OWNED_" .. tostring(Player.UserId)
local cacheEnv = (typeof(getgenv) == "function" and getgenv()) or _G
local function ownedFromCache()
    if cacheEnv[OWNED_KEY] == true then return true end
    if typeof(readfile) ~= "function" then return false end
    local ok, data = pcall(function()
        if typeof(isfile) == "function" and not isfile(OWNED_FILE) then return nil end
        return readfile(OWNED_FILE)
    end)
    if ok and data == "MAMORI_OWNED" then
        cacheEnv[OWNED_KEY] = true
        return true
    end
    return false
end
local function rememberMamori()
    cacheEnv[OWNED_KEY] = true
    if typeof(writefile) == "function" then
        pcall(function() writefile(OWNED_FILE, "MAMORI_OWNED") end)
    end
end

-- CEK DISPLAY NAME MAMORI DI SELURUH SLOT
local function hasMamori()
    if targetFound or ownedFromCache() then
        return true
    end

    local found = false

    pcall(function()
        if not TeamSel then
            TeamSel = Knit.GetController("TeamSelectionController")
        end

        local styles = TeamSel
            and TeamSel.Styles
            and TeamSel.Styles:get()

        if type(styles) == "table" and StyleMod then
            for _, id in pairs(styles) do
                local okDef, def = pcall(function() return StyleMod:Get(id) end)
                if okDef and def and type(def.DisplayName) == "string"
                    and def.DisplayName:match("^%s*(.-)%s*$"):lower() == "mamori" then
                    found = true
                    break
                end
            end
        end
    end)

    if found then
        targetFound = true
        rememberMamori()
    end

    return found
end

local function getRollRF()
    local idx = RS:FindFirstChild("Packages")
        and RS.Packages:FindFirstChild("_Index")

    if idx then
        for _, f in ipairs(idx:GetChildren()) do
            local k = f:FindFirstChild("knit")

            if k and k:FindFirstChild("Services") then
                local svc = k.Services:FindFirstChild("StyleService")

                if svc
                    and svc:FindFirstChild("RF")
                    and svc.RF:FindFirstChild("Roll") then

                    return svc.RF.Roll
                end
            end
        end
    end

    return nil
end

local rollRF = getRollRF()

local function checkRes(a, b)
    local ids = {}

    local function col(t)
        if type(t) ~= "table" then
            return
        end

        if t.Id or t.Style or t.Rarity then
            table.insert(ids, t.Id or t.Style)
        else
            for _, v in ipairs(t) do
                if type(v) == "string" then
                    table.insert(ids, v)
                end
            end
        end
    end

    col(b)
    col(a)

    local id = ids[#ids]

    if id and StyleMod then
        local ok, d = pcall(function()
            return StyleMod:Get(id)
        end)

        if ok and d then
            return d.Rarity, id
        end
    end

    return nil, id
end

local function doRoll(isLucky)
    if rollRF then
        local ok, a, b = pcall(
            rollRF.InvokeServer,
            rollRF,
            isLucky
        )

        return ok, a, b
    else
        local ok, a, b = pcall(function()
            local r1, r2 = StyleService:Roll(isLucky):expect()
            return r1, r2
        end)

        return ok, a, b
    end
end

-- ROLLBACK DENGAN PENGAMAN
local function doRollback()

    if targetFound or hasMamori() then
        stopSuccess()
        return
    end

    -- Tunggu sinkronisasi data style
    for i = 1, 5 do
        task.wait(0.3)

        if targetFound or hasMamori() then
            stopSuccess()
            return
        end
    end

    if targetFound or hasMamori() then
        stopSuccess()
        return
    end

    print("[AutoSpin] Mamori tidak terdeteksi -> rollback")
    mamoriBudget.markRollback()

    local ok = pcall(function()
        local r = RS
            :WaitForChild("Packages")
            :WaitForChild("_Index")
            :WaitForChild("sleitnick_knit@1.7.0")
            :WaitForChild("knit")
            :WaitForChild("Services")
            :WaitForChild("SettingsService")
            :WaitForChild("RF")
            :WaitForChild("UpdateKeybind")

        r:InvokeServer(
            "MouseButton1",
            true,
            "Spike\xE2\x80\x8B\x8F"
        )

        TeleportService:TeleportToPlaceInstance(
            game.PlaceId,
            game.JobId
        )
    end)

    if not ok then
        Player:Kick("Rollback gagal")
    end
end

-- MAIN LOOP
task.spawn(function()

    while running do

        -- CEK TARGET SETIAP LOOP
        if hasMamori() then
            stopSuccess()
            break
        end

        local spins, lucky = 0, 0

        pcall(function()
            spins = StyleCtrl.Spins
                and StyleCtrl.Spins:get()
                or 0
        end)

        pcall(function()
            lucky = StyleCtrl.LuckySpins
                and StyleCtrl.LuckySpins:get()
                or 0
        end)
        observeMamoriLucky(lucky)

        if stopMamoriIfNoSafeSlot() then break end
        if mamoriBudget.capped() then
            print("[AutoSpin] BATAS 30 Lucky Mamori tercapai -> rollback")
            saveFlag()
            doRollback()
            running = false
            break
        end
        if mamoriLuckyPendingBefore then
            if os.clock() - lastMamoriPendingLog > 6 then
                lastMamoriPendingLog = os.clock()
                warn("[AutoSpin] Menunggu saldo Lucky Mamori berkurang;"
                    .. " spin baru ditahan agar tidak melebihi 30")
            end
            task.wait(0.5)
            continue
        end

        if spins <= 0 and lucky <= 0 then
            print("[AutoSpin] Semua spin habis -> STOP")
            running = false
            clearFlag()
            break
        end

        -- Satu pemeriksaan lokal terakhir sebelum menggunakan Lucky Style Spins.
        if ownedFromCache() then
            stopSuccess()
            break
        end
        local ok, a, b = doRoll(lucky > 0)

        if not ok then
            warn("[AutoSpin] Roll ERROR: " .. tostring(a))
            task.wait(1)

        elseif a == false then
            print("[AutoSpin] Server menolak roll")
            task.wait(0.5)

        else
            local rar, id = checkRes(a, b)

            print("[AutoSpin] Rarity:", rar, "ID:", id)

            -- PRIORITAS UTAMA: MAMORI
            local resultIsMamori = false
            if id and StyleMod then
                local okDef, def = pcall(function() return StyleMod:Get(id) end)
                resultIsMamori = okDef and def
                    and type(def.DisplayName) == "string"
                    and def.DisplayName:match("^%s*(.-)%s*$"):lower() == "mamori"
            end
            if resultIsMamori then
                stopSuccess()
                break
            end

            -- CEK INVENTORY SETELAH SPIN
            if hasMamori() then
                stopSuccess()
                break
            end

            if lucky > 0 then
                mamoriLuckyPendingBefore = lucky
                -- Jangan mengirim roll Lucky berikutnya sebelum saldo
                -- mengonfirmasi pengurangan roll yang baru diterima.
                for _ = 1, 25 do
                    local afterValue
                    pcall(function() afterValue = StyleCtrl.LuckySpins:get() end)
                    observeMamoriLucky(afterValue)
                    if not mamoriLuckyPendingBefore then break end
                    if hasMamori() then
                        stopSuccess()
                        return
                    end
                    task.wait(0.2)
                end
                if mamoriLuckyPendingBefore then
                    warn("[AutoSpin] Roll Lucky Mamori diterima tetapi saldo belum"
                        .. " turun; menunggu sinkronisasi, tidak spin lagi")
                    task.wait(0.5)
                    continue
                end
            end

            local luckyAfter = 0

            pcall(function()
                luckyAfter = StyleCtrl.LuckySpins
                    and StyleCtrl.LuckySpins:get()
                    or 0
            end)
            observeMamoriLucky(luckyAfter)

            if luckyAfter <= 0 or mamoriBudget.capped() then

                -- CEK SEBELUM ROLLBACK
                if hasMamori() then
                    stopSuccess()
                    break
                end

                print(mamoriBudget.capped()
                    and "[AutoSpin] Batas 30 Lucky tercapai, memeriksa Mamori"
                    or "[AutoSpin] Lucky habis, memeriksa Mamori")

                -- TUNGGU UPDATE DATA
                task.wait(1)

                if hasMamori() then
                    stopSuccess()
                    break
                end

                -- MAMORI BELUM TERDETEKSI
                if stopMamoriIfNoSafeSlot() then break end
                saveFlag()
                doRollback()
                running = false
                break
            end
        end

        task.wait(0.55)
    end
end)


-- =====================================================================
-- TAMBAHAN TERPISAH: PEMANTAU ROLLBACK
-- Tidak mengedit doRollback(), doRoll(), atau MAIN LOOP asli.
-- =====================================================================
do
    local Log = game:GetService("LogService")
    local teleportCalled = false
    local rollbackMentioned = false
    local hasRolledThisSession = false
    local fallbackUsed = false
    local teleportRetries = 0
    local teleportStarted = false
    local rollErrors = 0

    -- Event ini hanya mengamati keluaran skrip inti. Jika log tidak keluar
    -- melalui LogService pada executor tertentu, fallback log tidak aktif.
    Log.MessageOut:Connect(function(message)
        if type(message) ~= "string" then return end
        if message:find("[AutoSpin] Mamori SUDAH DIDAPATKAN", 1, true) then
            fallbackUsed = true
            return
        end
        if message:find("[AutoSpin] Rarity:", 1, true) then
            hasRolledThisSession = true
            rollErrors = 0
        end
        if message:find("[AutoSpin] Mamori tidak terdeteksi -> rollback", 1, true) then
            rollbackMentioned = true
            teleportCalled = true
            task.delay(12, function()
                if not teleportStarted and not hasMamori() then
                    warn("[ROLLBACK GUARD] Rollback diminta, tetapi teleport belum terkonfirmasi. Periksa UpdateKeybind/TeleportService.")
                end
            end)
        end
        if message:find("[AutoSpin] Roll ERROR:", 1, true)
            or message:find("[AutoSpin] Server menolak roll", 1, true) then
            rollErrors = rollErrors + 1
            if rollErrors == 8 then
                warn("[ROLLBACK GUARD] Roll gagal 8 kali berturut-turut; skrip asli belum mencapai cabang rollback")
            end
        end
        if message:find("[AutoSpin] Semua spin habis -> STOP", 1, true)
            and hasRolledThisSession and not fallbackUsed and not rollbackMentioned then
            -- Skrip asli menghentikan diri pada kondisi semua spin 0.
            -- Fallback hanya kalau setidaknya ada roll berhasil sebelumnya.
            fallbackUsed = true
            task.spawn(function()
                task.wait(0.6)
                if hasMamori() then return end
                local stable = true
                for _ = 1, 4 do
                    local ok, spins, lucky = pcall(function()
                        return StyleCtrl.Spins:get(), StyleCtrl.LuckySpins:get()
                    end)
                    if not ok or type(spins) ~= "number" or type(lucky) ~= "number"
                        or spins > 0 or lucky > 0 then
                        stable = false
                        break
                    end
                    task.wait(0.4)
                end
                if not stable or hasMamori() or rollbackMentioned then
                    warn("[ROLLBACK GUARD] Stop 0-spin terjadi, tetapi data tidak stabil/target sudah ada; fallback dibatalkan")
                    return
                end
                warn("[ROLLBACK GUARD] Semua spin habis setelah roll; mencoba fungsi rollback ASLI sekali")
                saveFlag()
                local ok, err = pcall(doRollback)
                if not ok then
                    warn("[ROLLBACK GUARD] doRollback asli error:", tostring(err))
                end
            end)
        end
    end)

    -- Mengulang permintaan teleport hanya jika Roblox melaporkan gagal,
    -- dan hanya setelah cabang rollback asli benar-benar dipanggil.
    pcall(function()
        Player.OnTeleport:Connect(function(state)
            if state == Enum.TeleportState.Started or state == Enum.TeleportState.InProgress then
                teleportStarted = true
            end
        end)
    end)
    TeleportService.TeleportInitFailed:Connect(function(player, result, errorMessage)
        if player ~= Player or not rollbackMentioned or hasMamori() then return end
        warn("[ROLLBACK GUARD] Teleport gagal:", tostring(result), tostring(errorMessage))
        if teleportRetries >= 2 then
            warn("[ROLLBACK GUARD] Batas 2 percobaan ulang teleport tercapai")
            return
        end
        teleportRetries = teleportRetries + 1
        task.delay(2, function()
            if hasMamori() then return end
            local ok, err = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId)
            end)
            if not ok then
                warn("[ROLLBACK GUARD] Retry teleport error:", tostring(err))
            end
        end)
    end)
end

    end
    -- PENGAMAN PASCA-REJOIN (FIX): tunggu data StyleController TERBACA,
    -- bukan menunggu jumlah spin harus lebih dari nol.
    -- Nilai 0 yang valid harus diteruskan ke MAIN LOOP asli,
    -- yang menentukan sendiri apakah spin habis.
    local function waitForSpinData()
        __mamoriTrace("waitForSpinData() DIMULAI")
        local deadline = os.clock() + 30
        local lastStatus = ""
        local lastSpins, lastLucky = nil, nil
        local stableCount = 0
        while os.clock() < deadline do
            local ok, spins, lucky = pcall(function()
                local packages = RS:FindFirstChild("Packages")
                local knitModule = packages and packages:FindFirstChild("Knit")
                if not knitModule then return nil, nil end
                local knit = require(knitModule)
                local ctrl = knit.GetController("StyleController")
                if not ctrl then return nil, nil end
                local s = ctrl.Spins and ctrl.Spins:get()
                local l = ctrl.LuckySpins and ctrl.LuckySpins:get()
                return s, l
            end)
            if ok and type(spins) == "number" and type(lucky) == "number"
                and spins >= 0 and lucky >= 0 then
                if spins == lastSpins and lucky == lastLucky then
                    stableCount = stableCount + 1
                else
                    stableCount = 1
                end
                lastSpins, lastLucky = spins, lucky
                if stableCount >= 2 then
                    print("[FINAL] Data spin terbaca:", spins, "Lucky:", lucky)
                    __mamoriTrace("SPIN DATA SIAP spins="..tostring(spins).." lucky="..tostring(lucky))
                    return true
                end
                lastStatus = "menunggu angka stabil"
            else
                stableCount = 0
                lastStatus = "StyleController/data belum siap"
            end
            task.wait(0.5)
        end
        warn("[FINAL] Tidak dapat membaca data spin dalam 30 detik: " .. lastStatus)
        if typeof(writefile) == "function" then
            pcall(function()
                writefile("VL_Mamori_Startup_Status.txt", "FAIL: " .. lastStatus)
            end)
        end
        return false
    end

    if waitForSpinData() then
        if typeof(writefile) == "function" then
            pcall(function()
                writefile("VL_Mamori_Startup_Status.txt", "READY: Memulai AutoSpin. Jika spin dan lucky 0, MAIN LOOP asli akan STOP.")
            end)
        end
        __mamoriTrace("MEMANGGIL runMamori() SEKARANG")
        runMamori()
        __mamoriTrace("runMamori() KEMBALI")
    else
        __mamoriTrace("GAGAL waitForSpinData() -> AutoSpin tidak dimulai")
    end
end
